Why Instant Updates?

Traditional app updates require users to download from the App Store or Play Store, which can take days for approval. Code push solutions allow you to push JavaScript bundle updates directly to users' devices instantly.

Key LimitationCode push can only update JavaScript code, not native modules. Native changes still require a full app store release.

// Manual update check
import codePush from 'appcenter-codepush';

async function checkForUpdate() {
  try {
    const update = await codePush.checkForUpdate();
    if (update) {
      console.log('Update available:', update.description);
      // Optionally show update dialog
      showUpdateDialog(update);
    }
  } catch (error) {
    console.error('Error checking for update:', error);
  }
}

// Download and apply update
async function applyUpdate() {
  try {
    await codePush.sync(
      {
        installMode: codePush.InstallMode.IMMEDIATE,
        updateDialog: true,
      },
      (status) => {
        switch (status) {
          case codePush.SyncStatus.DOWNLOADING_PACKAGE:
            console.log('Downloading update...');
            break;
          case codePush.SyncStatus.INSTALLING_UPDATE:
            console.log('Installing update...');
            break;
          case codePush.SyncStatus.UP_TO_DATE:
            console.log('App is up to date');
            break;
        }
      }
    );
  } catch (error) {
    console.error('Error applying update:', error);
  }
}

// In-app update handling
import * as Updates from 'expo-updates';

async function checkForUpdates() {
  try {
    if (Updates.isAvailable) {
      const { isRollback } = await Updates.fetchUpdateAsync();
      if (isRollback) {
        console.log('Rolled back to previous version');
      } else {
        await Updates.reloadAsync();
      }
    }
  } catch (error) {
    console.error('Error checking for updates:', error);
  }
}

// Custom update channel strategy
async function checkForChannelUpdate(channel) {
  try {
    const update = await Updates.checkForUpdateAsync();
    if (update) {
      console.log(`Update available for channel: ${channel}`);
      await Updates.fetchUpdateAsync();
      await Updates.reloadAsync();
    }
  } catch (error) {
    console.error('Update check failed:', error);
  }
}

// Rollback handling
Updates.addListener((event) => {
  if (event.type === Updates.UpdateEventType.ROLLBACK_RESTARTED) {
    console.log('Update rolled back, restarting app');
    // Log rollback for monitoring
    logRollbackEvent(event);
  }
});

Deployment Strategies

// Staged rollout strategy
const DEPLOYMENT_CONFIG = {
  canary: {
    percentage: 5,
    duration: '24h',
    rollbackThreshold: 0.01,
  },
  beta: {
    percentage: 25,
    duration: '48h',
    rollbackThreshold: 0.005,
  },
  production: {
    percentage: 100,
    duration: 'none',
    rollbackThreshold: 0.001,
  },
};

// Implement staged rollout
async function deployUpdate(update, stage) {
  const config = DEPLOYMENT_CONFIG[stage];
  
  // Monitor error rates
  const errorRate = await monitorErrorRate(update.id);
  
  if (errorRate > config.rollbackThreshold) {
    console.log(`Error rate ${errorRate} exceeds threshold, rolling back`);
    await rollbackUpdate(update.id);
    return false;
  }
  
  // Deploy to percentage of users
  await deployToPercentage(update.id, config.percentage);
  
  // Set up monitoring for next stage
  if (stage !== 'production') {
    scheduleStageProgression(update.id, stage);
  }
  
  return true;
}

// Rollback mechanism
async function rollbackUpdate(updateId) {
  try {
    // Immediately push previous stable version
    await codePush.rollback();
    console.log('Rollback completed successfully');
    
    // Notify monitoring systems
    await notifyRollback(updateId);
  } catch (error) {
    console.error('Rollback failed:', error);
    // Fallback: force app restart to previous bundle
    await forceRestartPreviousVersion();
  }
}

// Version management
const versionManager = {
  getCurrentVersion: () => {
    return {
      jsBundle: Updates.channel,
      nativeVersion: appVersion,
      codePushVersion: codePush.getUpdateMetadata(),
    };
  },
  
  getVersionHistory: async () => {
    const history = await AsyncStorage.getItem('version_history');
    return history ? JSON.parse(history) : [];
  },
  
  trackUpdate: async (newVersion) => {
    const history = await versionManager.getVersionHistory();
    history.push({
      ...newVersion,
      timestamp: Date.now(),
      user: getCurrentUser(),
    });
    await AsyncStorage.setItem('version_history', JSON.stringify(history));
  },
};

Best Practices

  • Always test updates in staging before production
  • Implement monitoring and alerting for update failures
  • Keep native dependencies minimal to maximize code push coverage
  • Use feature flags to control new feature rollouts
  • Maintain rollback capability for critical issues
  • Version your updates properly for tracking

// 2. Create update service
// services/updateService.js
import * as Updates from 'expo-updates';
import AsyncStorage from '@react-native-async-storage/async-storage';

class UpdateService {
  static async checkForUpdate() {
    try {
      if (!Updates.isAvailable) return null;
      
      const update = await Updates.checkForUpdateAsync();
      return update;
    } catch (error) {
      console.error('Update check failed:', error);
      return null;
    }
  }

  static async applyUpdate(onProgress) {
    try {
      const update = await Updates.fetchUpdateAsync();
      
      if (onProgress) {
        onProgress({ status: 'downloading', progress: 100 });
      }
      
      if (update.isRollback) {
        await this.handleRollback();
      } else {
        await Updates.reloadAsync();
      }
      
      return true;
    } catch (error) {
      console.error('Update failed:', error);
      return false;
    }
  }

  static async handleRollback() {
    await AsyncStorage.setItem('lastRollback', Date.now().toString());
    console.log('Update rolled back to previous version');
  }

  static async getVersionHistory() {
    const history = await AsyncStorage.getItem('versionHistory');
    return history ? JSON.parse(history) : [];
  }
}

// 3. Create update hook
// hooks/useUpdate.js
import { useState, useEffect } from 'react';
import UpdateService from '../services/updateService';

export function useUpdate() {
  const [updateAvailable, setUpdateAvailable] = useState(false);
  const [isUpdating, setIsUpdating] = useState(false);
  const [error, setError] = useState(null);

  useEffect(() => {
    checkForUpdates();
  }, []);

  const checkForUpdates = async () => {
    const update = await UpdateService.checkForUpdate();
    setUpdateAvailable(!!update);
  };

  const applyUpdate = async () => {
    setIsUpdating(true);
    setError(null);
    
    try {
      await UpdateService.applyUpdate((progress) => {
        console.log('Update progress:', progress);
      });
    } catch (err) {
      setError(err.message);
    } finally {
      setIsUpdating(false);
    }
  };

  return {
    updateAvailable,
    isUpdating,
    error,
    checkForUpdates,
    applyUpdate,
  };
}

// 4. Usage in App component
function App() {
  const { updateAvailable, applyUpdate } = useUpdate();

  useEffect(() => {
    if (updateAvailable) {
      showUpdateDialog();
    }
  }, [updateAvailable]);

  return (
    <View style={{ flex: 1 }}>
      {/* Your app content */}
      {updateAvailable && (
        <UpdateBanner onApply={applyUpdate} />
      )}
    </View>
  );
}
