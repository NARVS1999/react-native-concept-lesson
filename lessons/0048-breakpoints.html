Lesson 48: Breakpoints

Phase 6 — Design Concepts

Lesson Goal By the end of this lesson, you will understand how to define breakpoints, distinguish between phone/tablet/desktop layouts, create a custom useBreakpoint hook, and build adaptive layouts based on screen size.

Why Breakpoints?

Breakpoints are predefined screen width thresholds that trigger different layout behaviors. Instead of checking width > 600 throughout your code, you define named breakpoints like isTablet or isDesktop and use them consistently.

This approach is similar to CSS media queries but implemented in JavaScript. Common breakpoints for React Native are:

  • Phone (small): 0–374px
  • Phone (regular): 375–599px
  • Tablet: 600–899px
  • Desktop: 900px+

Defining Breakpoints

Create a centralized breakpoint configuration file that defines your screen size thresholds.

// theme/breakpoints.js

// Standard device breakpoints
export const breakpoints = {
  phoneSmall: 0,
  phone: 375,
  phoneLarge: 414,
  tablet: 600,
  tabletLarge: 768,
  desktop: 900,
  desktopLarge: 1200,
};

// Shorthand checks
export const screens = {
  isPhoneSmall: (width) => width < breakpoints.phone,
  isPhone: (width) => width >= breakpoints.phone && width < breakpoints.tablet,
  isPhoneLarge: (width) => width >= breakpoints.phoneLarge && width < breakpoints.tablet,
  isTablet: (width) => width >= breakpoints.tablet && width < breakpoints.desktop,
  isTabletLarge: (width) => width >= breakpoints.tabletLarge && width < breakpoints.desktop,
  isDesktop: (width) => width >= breakpoints.desktop,
  isDesktopLarge: (width) => width >= breakpoints.desktopLarge,
};

// Column counts per breakpoint
export const gridColumns = {
  phoneSmall: 1,
  phone: 2,
  phoneLarge: 2,
  tablet: 3,
  tabletLarge: 4,
  desktop: 4,
  desktopLarge: 6,
};

Building a useBreakpoint Hook

A custom hook that encapsulates breakpoint logic and provides easy-to-use boolean flags.

// hooks/useBreakpoint.js
import { useWindowDimensions } from 'react-native';
import { breakpoints } from '../theme/breakpoints';

export default function useBreakpoint() {
  const { width, height } = useWindowDimensions();

  const breakpoint = {
    // Boolean flags
    isPhoneSmall: width < breakpoints.phone,
    isPhone: width >= breakpoints.phone && width < breakpoints.tablet,
    isTablet: width >= breakpoints.tablet && width < breakpoints.desktop,
    isDesktop: width >= breakpoints.desktop,
    isDesktopLarge: width >= breakpoints.desktopLarge,

    // Current breakpoint name
    current: getCurrentBreakpoint(width),

    // Screen dimensions
    width,
    height,
  };

  return breakpoint;
}

function getCurrentBreakpoint(width) {
  if (width >= breakpoints.desktopLarge) return 'desktopLarge';
  if (width >= breakpoints.desktop) return 'desktop';
  if (width >= breakpoints.tabletLarge) return 'tabletLarge';
  if (width >= breakpoints.tablet) return 'tablet';
  if (width >= breakpoints.phoneLarge) return 'phoneLarge';
  if (width >= breakpoints.phone) return 'phone';
  return 'phoneSmall';
}

// Alternative: more granular hook
export function useBreakpointValue(values) {
  const { width } = useWindowDimensions();

  if (width >= breakpoints.desktop && values.desktop !== undefined) {
    return values.desktop;
  }
  if (width >= breakpoints.tablet && values.tablet !== undefined) {
    return values.tablet;
  }
  return values.phone ?? values.default;
}

// Usage example
function ResponsiveComponent() {
  const columns = useBreakpointValue({
    phone: 1,
    tablet: 2,
    desktop: 3,
    default: 1,
  });

  return (
    <View style={{ flex: 1 }}>
      <Text>Showing {columns} columns</Text>
    </View>
  );
}

Adaptive Layouts Based on Screen Size

Use the breakpoint hook to create layouts that adapt dramatically between phone, tablet, and desktop.

// components/AdaptiveLayout.js
import { View, Text, StyleSheet } from 'react-native';
import useBreakpoint from '../hooks/useBreakpoint';

export default function AdaptiveLayout({ children }) {
  const { isPhone, isTablet, isDesktop, width } = useBreakpoint();

  // Sidebar visibility
  const showSidebar = isTablet || isDesktop;
  const sidebarWidth = isDesktop ? 280 : 240;

  // Content padding
  const contentPadding = isPhone ? 16 : isTablet ? 24 : 32;

  // Font scaling
  const headerSize = isPhone ? 20 : isTablet ? 24 : 28;

  return (
    <View style={styles.container}>
      {showSidebar && (
        <View style={[styles.sidebar, { width: sidebarWidth }]}>
          <Text style={styles.sidebarTitle}>Navigation</Text>
          <View style={styles.navItem}>
            <Text style={styles.navText}>Dashboard</Text>
          </View>
          <View style={styles.navItem}>
            <Text style={styles.navText}>Settings</Text>
          </View>
          <View style={styles.navItem}>
            <Text style={styles.navText}>Profile</Text>
          </View>
        </View>
      )}
      <View style={[styles.main, { padding: contentPadding }]}>
        <Text style={[styles.header, { fontSize: headerSize }]}>
          Adaptive Content
        </Text>
        <Text style={styles.info}>
          Screen: {width}px | {isPhone ? 'Phone' : isTablet ? 'Tablet' : 'Desktop'}
        </Text>
        <View style={styles.contentGrid}>
          {children}
        </View>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, flexDirection: 'row' },
  sidebar: {
    backgroundColor: '#1f2937',
    padding: 16,
  },
  sidebarTitle: { color: '#fff', fontSize: 18, fontWeight: 'bold', marginBottom: 16 },
  navItem: { paddingVertical: 12, paddingHorizontal: 8 },
  navText: { color: '#d1d5db', fontSize: 16 },
  main: { flex: 1, backgroundColor: '#f9fafb' },
  header: { fontWeight: 'bold', marginBottom: 8 },
  info: { color: '#6b7280', marginBottom: 16 },
  contentGrid: { flex: 1 },
});

Responsive Grid System

Build a grid system that automatically adjusts columns based on breakpoints.

// components/ResponsiveGrid.js
import { View, Text, useWindowDimensions, StyleSheet } from 'react-native';
import { breakpoints } from '../theme/breakpoints';

function getGridColumns(width) {
  if (width >= breakpoints.desktop) return 4;
  if (width >= breakpoints.tablet) return 3;
  if (width >= breakpoints.phoneLarge) return 2;
  return 1;
}

function getGap(width) {
  if (width >= breakpoints.desktop) return 16;
  if (width >= breakpoints.tablet) return 12;
  return 8;
}

export default function ResponsiveGrid({ items, renderItem }) {
  const { width } = useWindowDimensions();
  const columns = getGridColumns(width);
  const gap = getGap(width);
  const itemWidth = (width - 32 - gap * (columns - 1)) / columns;

  // Split items into rows
  const rows = [];
  for (let i = 0; i < items.length; i += columns) {
    rows.push(items.slice(i, i + columns));
  }

  return (
    <View style={styles.container}>
      {rows.map((row, rowIndex) => (
        <View key={rowIndex} style={[styles.row, { gap }]}>
          {row.map((item, colIndex) => (
            <View key={colIndex} style={{ width: itemWidth }}>
              {renderItem(item)}
            </View>
          ))}
          {/* Fill empty spaces in last row */}
          {row.length < columns &&
            Array.from({ length: columns - row.length }).map((_, i) => (
              <View key={`empty-${i}`} style={{ width: itemWidth }} />
            ))}
        </View>
      ))}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { padding: 16 },
  row: { flexDirection: 'row', marginBottom: 8 },
});

Mini-Project: Build an Adaptive Layout with 3 Breakpoints

Create a layout that changes dramatically across phone, tablet, and desktop:

// hooks/useBreakpoint.js
import { useWindowDimensions } from 'react-native';

const BREAKPOINTS = {
  phone: 0,
  tablet: 600,
  desktop: 900,
};

export default function useBreakpoint() {
  const { width, height } = useWindowDimensions();

  return {
    width,
    height,
    isPhone: width < BREAKPOINTS.tablet,
    isTablet: width >= BREAKPOINTS.tablet && width < BREAKPOINTS.desktop,
    isDesktop: width >= BREAKPOINTS.desktop,
    current: width >= BREAKPOINTS.desktop
      ? 'desktop'
      : width >= BREAKPOINTS.tablet
        ? 'tablet'
        : 'phone',
  };
}

// screens/AdaptiveScreen.js
import { View, Text, ScrollView, StyleSheet } from 'react-native';
import useBreakpoint from '../hooks/useBreakpoint';

export default function AdaptiveScreen() {
  const { isPhone, isTablet, isDesktop, current } = useBreakpoint();

  return (
    <View style={styles.container}>
      {/* Navigation */}
      {isDesktop && (
        <View style={styles.sidebar}>
          <Text style={styles.navTitle}>Menu</Text>
          {['Home', 'Profile', 'Settings'].map(item => (
            <Text key={item} style={styles.navItem}>{item}</Text>
          ))}
        </View>
      )}

      {/* Main content */}
      <ScrollView style={styles.main}>
        <Text style={styles.title}>
          {isPhone ? 'Phone View' : isTablet ? 'Tablet View' : 'Desktop View'}
        </Text>
        <Text style={styles.subtitle}>Current breakpoint: {current}</Text>

        {/* Cards grid */}
        <View style={[
          styles.grid,
          isTablet && styles.gridTablet,
          isDesktop && styles.gridDesktop,
        ]}>
          {[1,2,3,4,5,6].map(i => (
            <View key={i} style={styles.card}>
              <Text style={styles.cardText}>Card {i}</Text>
            </View>
          ))}
        </View>
      </ScrollView>

      {/* Bottom tabs on phone only */}
      {isPhone && (
        <View style={styles.tabBar}>
          <Text style={styles.tab}>Home</Text>
          <Text style={styles.tab}>Search</Text>
          <Text style={styles.tab}>Profile</Text>
        </View>
      )}
    </View>
  );
}

Summary

  • Breakpoints are named screen width thresholds that trigger layout changes
  • Common ranges: phone (0–599px), tablet (600–899px), desktop (900px+)
  • A useBreakpoint hook provides boolean flags (isPhone, isTablet, isDesktop) for easy conditional rendering
  • Centralize breakpoint values in a config file for consistency across the app
  • Use breakpoint-based grid systems to automatically adjust column counts
  • Test on real devices or simulators to verify your breakpoints work correctly
