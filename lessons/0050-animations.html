Lesson 50: Animations

Phase 6 — Design Concepts

Lesson Goal By the end of this lesson, you will understand the Animated API basics, Animated.Value, timing/spring/decay animations, Animated.View, and composition methods like parallel, sequence, and stagger.

Why Animations Matter

Animations make your app feel responsive and polished. They provide visual feedback, guide user attention, and create delightful interactions. React Native's Animated API allows you to create smooth 60fps animations that run on the native thread.

Animated API Basics

The Animated API works by declaring animated values and binding them to component styles. When the values change, React Native efficiently updates only the specific styles that changed, without re-rendering the component tree.

import React, { useRef } from 'react';
import { Animated, View, Text, Pressable, StyleSheet } from 'react-native';

function BasicAnimation() {
  const fadeAnim = useRef(new Animated.Value(0)).current;

  const fadeIn = () => {
    Animated.timing(fadeAnim, {
      toValue: 1,
      duration: 1000,
      useNativeDriver: true,
    }).start();
  };

  return (
    <View style={styles.container}>
      <Animated.View style={[styles.box, { opacity: fadeAnim }]}>
        <Text style={styles.boxText}>Fade In</Text>
      </Animated.View>
      <Pressable onPress={fadeIn} style={styles.button}>
        <Text style={styles.buttonText}>Start Animation</Text>
      </Pressable>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  box: { width: 150, height: 150, backgroundColor: '#3b82f6', borderRadius: 12, justifyContent: 'center', alignItems: 'center' },
  boxText: { color: '#fff', fontWeight: 'bold' },
  button: { marginTop: 20, padding: 12, backgroundColor: '#10b981', borderRadius: 8 },
  buttonText: { color: '#fff', fontWeight: '600' },
});

Animated.Value

Animated.Value is a wrapper around a number that can track its changes over time. You create it with useRef to persist it across re-renders.

import React, { useRef } from 'react';
import { Animated, View, StyleSheet } from 'react-native';

function AnimatedValueDemo() {
  // Create animated values
  const scaleValue = useRef(new Animated.Value(1)).current;
  const rotateValue = useRef(new Animated.Value(0)).current;
  const translateValue = useRef(new Animated.Value(0)).current;

  // Interpolate values for complex animations
  const rotate = rotateValue.interpolate({
    inputRange: [0, 1],
    outputRange: ['0deg', '360deg'],
  });

  const backgroundColor = translateValue.interpolate({
    inputRange: [0, 100],
    outputRange: ['#3b82f6', '#ef4444'],
  });

  return (
    <View style={styles.container}>
      <Animated.View
        style={[
          styles.box,
          {
            transform: [
              { scale: scaleValue },
              { translateX: translateValue },
              { rotate },
            ],
            backgroundColor,
          },
        ]}
      />
    </View>
  );
}

Timing Animations

Animated.timing animates a value from its current state to a target value over a specified duration with an easing function.

import React, { useRef } from 'react';
import { Animated, View, Pressable, Text, StyleSheet } from 'react-native';

function TimingAnimation() {
  const opacity = useRef(new Animated.Value(0)).current;
  const translateY = useRef(new Animated.Value(50)).current;
  const scale = useRef(new Animated.Value(0.5)).current;

  const animateIn = () => {
    Animated.parallel([
      Animated.timing(opacity, {
        toValue: 1,
        duration: 600,
        useNativeDriver: true,
      }),
      Animated.timing(translateY, {
        toValue: 0,
        duration: 600,
        useNativeDriver: true,
      }),
      Animated.timing(scale, {
        toValue: 1,
        duration: 600,
        useNativeDriver: true,
      }),
    ]).start();
  };

  const animateOut = () => {
    Animated.parallel([
      Animated.timing(opacity, { toValue: 0, duration: 400, useNativeDriver: true }),
      Animated.timing(translateY, { toValue: -50, duration: 400, useNativeDriver: true }),
      Animated.timing(scale, { toValue: 0.5, duration: 400, useNativeDriver: true }),
    ]).start();
  };

  return (
    <View style={styles.container}>
      <Animated.View
        style={[
          styles.card,
          { opacity, transform: [{ translateY }, { scale }] },
        ]}
      >
        <Text style={styles.cardTitle}>Animated Card</Text>
        <Text style={styles.cardText}>This card animates in and out</Text>
      </Animated.View>
      <View style={styles.buttons}>
        <Pressable onPress={animateIn} style={styles.button}>
          <Text style={styles.buttonText}>Show</Text>
        </Pressable>
        <Pressable onPress={animateOut} style={[styles.button, styles.buttonSecondary]}>
          <Text style={[styles.buttonText, styles.buttonSecondaryText]}>Hide</Text>
        </Pressable>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  card: { width: 280, padding: 24, backgroundColor: '#fff', borderRadius: 16, elevation: 8, shadowColor: '#000', shadowOffset: { width: 0, height: 4 }, shadowOpacity: 0.15, shadowRadius: 8 },
  cardTitle: { fontSize: 20, fontWeight: 'bold', marginBottom: 8 },
  cardText: { fontSize: 14, color: '#666' },
  buttons: { flexDirection: 'row', gap: 12, marginTop: 24 },
  button: { paddingVertical: 10, paddingHorizontal: 20, backgroundColor: '#3b82f6', borderRadius: 8 },
  buttonText: { color: '#fff', fontWeight: '600' },
  buttonSecondary: { backgroundColor: 'transparent', borderWidth: 1, borderColor: '#3b82f6' },
  buttonSecondaryText: { color: '#3b82f6' },
});

Spring Animations

Animated.spring creates a spring-based animation that simulates real physics. The animation overshoots slightly before settling, giving a natural feel.

import React, { useRef } from 'react';
import { Animated, View, Pressable, Text, StyleSheet } from 'react-native';

function SpringAnimation() {
  const scale = useRef(new Animated.Value(1)).current;

  const bounce = () => {
    // Reset and animate with spring
    scale.setValue(0.8);
    Animated.spring(scale, {
      toValue: 1,
      friction: 3,       // Lower = more bounce
      tension: 40,       // Higher = faster
      useNativeDriver: true,
    }).start();
  };

  const wobble = () => {
    Animated.sequence([
      Animated.spring(scale, { toValue: 1.2, friction: 2, useNativeDriver: true }),
      Animated.spring(scale, { toValue: 1, friction: 2, useNativeDriver: true }),
    ]).start();
  };

  return (
    <View style={styles.container}>
      <Animated.View style={[styles.ball, { transform: [{ scale }] }]} />
      <View style={styles.buttons}>
        <Pressable onPress={bounce} style={styles.button}>
          <Text style={styles.buttonText}>Bounce</Text>
        </Pressable>
        <Pressable onPress={wobble} style={styles.button}>
          <Text style={styles.buttonText}>Wobble</Text>
        </Pressable>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  ball: { width: 120, height: 120, borderRadius: 60, backgroundColor: '#8b5cf6' },
  buttons: { flexDirection: 'row', gap: 16, marginTop: 32 },
  button: { padding: 12, backgroundColor: '#3b82f6', borderRadius: 8 },
  buttonText: { color: '#fff', fontWeight: '600' },
});

Decay Animations

Animated.decay slows down a value from an initial velocity to rest. It's useful for scroll-based animations or gesture-driven interactions.

import React, { useRef } from 'react';
import { Animated, View, PanResponder, StyleSheet } from 'react-native';

function DecayAnimation() {
  const translateX = useRef(new Animated.Value(0)).current;
  const translateY = useRef(new Animated.Value(0)).current;

  const panResponder = PanResponder.create({
    onMoveShouldSetPanResponder: () => true,
    onPanResponderGrant: () => {
      translateX.setOffset(translateX._value);
      translateY.setOffset(translateY._value);
    },
    onPanResponderMove: Animated.event(
      [null, { dx: translateX, dy: translateY }],
      { useNativeDriver: false }
    ),
    onPanResponderRelease: (e, { vx, vy }) => {
      translateX.flattenOffset();
      translateY.flattenOffset();

      // Animate to rest with decay based on velocity
      Animated.decay(translateX, {
        velocity: vx,
        deceleration: 0.997,
        useNativeDriver: false,
      }).start();

      Animated.decay(translateY, {
        velocity: vy,
        deceleration: 0.997,
        useNativeDriver: false,
      }).start();
    },
  });

  return (
    <View style={styles.container}>
      <Animated.View
        style={[styles.circle, { transform: [{ translateX }, { translateY }] }]}
        {...panResponder.panHandlers}
      />
      <Text style={styles.hint}>Drag and release the circle</Text>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  circle: { width: 80, height: 80, borderRadius: 40, backgroundColor: '#ef4444' },
  hint: { marginTop: 24, color: '#666' },
});

Animated.View

Animated.View is a drop-in replacement for View that supports animated styles. Use it whenever you need to animate layout properties like opacity, transform, or backgroundColor.

import React, { useRef, useEffect } from 'react';
import { Animated, View, Text, StyleSheet } from 'react-native';

function AnimatedViewExample() {
  const fadeAnim = useRef(new Animated.Value(0)).current;
  const slideAnim = useRef(new Animated.Value(-100)).current;

  useEffect(() => {
    Animated.parallel([
      Animated.timing(fadeAnim, { toValue: 1, duration: 800, useNativeDriver: true }),
      Animated.timing(slideAnim, { toValue: 0, duration: 800, useNativeDriver: true }),
    ]).start();
  }, []);

  return (
    <Animated.View
      style={[
        styles.slideIn,
        {
          opacity: fadeAnim,
          transform: [{ translateY: slideAnim }],
        },
      ]}
    >
      <Text style={styles.text}>This content slides and fades in!</Text>
    </Animated.View>
  );
}

const styles = StyleSheet.create({
  slideIn: { padding: 24, backgroundColor: '#3b82f6', borderRadius: 12, margin: 16 },
  text: { color: '#fff', fontSize: 18, fontWeight: '600', textAlign: 'center' },
});

Parallel, Sequence, and Stagger

Compose multiple animations using parallel (simultaneous), sequence (one after another), and stagger (delayed starts).

import React, { useRef, useEffect } from 'react';
import { Animated, View, StyleSheet } from 'react-native';

// Parallel: Run animations simultaneously
function ParallelExample() {
  const opacity = useRef(new Animated.Value(0)).current;
  const scale = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    Animated.parallel([
      Animated.timing(opacity, { toValue: 1, duration: 500, useNativeDriver: true }),
      Animated.spring(scale, { toValue: 1, friction: 4, useNativeDriver: true }),
    ]).start();
  }, []);

  return (
    <Animated.View style={[styles.box, { opacity, transform: [{ scale }] }]} />
  );
}

// Sequence: Run animations one after another
function SequenceExample() {
  const color1 = useRef(new Animated.Value(0)).current;
  const color2 = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    Animated.sequence([
      Animated.timing(color1, { toValue: 1, duration: 500, useNativeDriver: false }),
      Animated.timing(color2, { toValue: 1, duration: 500, useNativeDriver: false }),
    ]).start();
  }, []);

  const bg = color1.interpolate({ inputRange: [0, 1], outputRange: ['#3b82f6', '#10b981'] });

  return <Animated.View style={[styles.box, { backgroundColor: bg }]} />;
}

// Stagger: Delay each animation by a time interval
function StaggerExample() {
  const items = [useRef(new Animated.Value(0)).current,
                 useRef(new Animated.Value(0)).current,
                 useRef(new Animated.Value(0)).current,
                 useRef(new Animated.Value(0)).current];

  useEffect(() => {
    Animated.stagger(150,
      items.map(item =>
        Animated.timing(item, { toValue: 1, duration: 400, useNativeDriver: true })
      )
    ).start();
  }, []);

  return (
    <View>
      {items.map((item, i) => (
        <Animated.View key={i} style={[styles.listItem, { opacity: item, transform: [{ translateY: item.interpolate({ inputRange: [0, 1], outputRange: [20, 0] }) }] }]}>
          <Text style={styles.listText}>Item {i + 1}</Text>
        </Animated.View>
      ))}
    </View>
  );
}

Mini-Project: Build a Fade-In Animation

Create a component that fades in with a slide-up effect:

// components/FadeInView.js
import { useEffect, useRef } from 'react';
import { Animated, StyleSheet } from 'react-native';

export default function FadeInView({ children, duration = 600, delay = 0 }) {
  const opacity = useRef(new Animated.Value(0)).current;
  const translateY = useRef(new Animated.Value(30)).current;

  useEffect(() => {
    Animated.parallel([
      Animated.timing(opacity, {
        toValue: 1,
        duration,
        delay,
        useNativeDriver: true,
      }),
      Animated.spring(translateY, {
        toValue: 0,
        delay,
        friction: 8,
        useNativeDriver: true,
      }),
    ]).start();
  }, []);

  return (
    <Animated.View
      style={[styles.container, { opacity, transform: [{ translateY }] }]}
    >
      {children}
    </Animated.View>
  );
}

const styles = StyleSheet.create({
  container: { overflow: 'hidden' },
});

// Usage
import { View, Text, StyleSheet } from 'react-native';
import FadeInView from './FadeInView';

export default function AnimatedList() {
  const items = ['First Item', 'Second Item', 'Third Item'];

  return (
    <View style={{ padding: 16 }}>
      {items.map((item, index) => (
        <FadeInView key={index} delay={index * 150}>
          <View style={styles.card}>
            <Text style={styles.cardText}>{item}</Text>
          </View>
        </FadeInView>
      ))}
    </View>
  );
}

  cardText: { color: '#fff', fontSize: 16, fontWeight: '600' },

Summary

  • Animated.Value wraps a number that can be animated over time
  • Animated.timing animates from current to target value with easing
  • Animated.spring creates physics-based animations with overshoot
  • Animated.decay slows a value from an initial velocity to rest
  • Animated.View is a drop-in replacement for View with animated style support
  • Use parallel, sequence, and stagger to compose multiple animations
  • Always set useNativeDriver: true for smooth 60fps animations when possible
