What is expo-file-system?

expo-file-system provides access to the local file system on the device. You can read, write, move, copy, and delete files, as well as download files from the internet.

Key Directories

  • documentDirectory — Persistent storage (backed up, survives app uninstall reinstall)
  • cacheDirectory — Temporary storage (can be cleared by the OS)
  • bundledResources — Read-only files from your app bundle

Installation

npx expo install expo-file-system

Reading and Writing Files

The two main operations are readAsStringAsync and writeAsStringAsync.

// utils/fileUtils.js
import * as FileSystem from 'expo-file-system';

const { documentDirectory, cacheDirectory } = FileSystem;

// Build file paths
const getFilePath = (filename) => `${documentDirectory}${filename}`;

// WRITE - Save text to a file
export async function saveToFile(filename, content) {
  const filePath = getFilePath(filename);
  try {
    await FileSystem.writeAsStringAsync(filePath, content, {
      encoding: FileSystem.EncodingType.UTF8,
    });
    console.log('File saved to:', filePath);
    return filePath;
  } catch (error) {
    console.error('Error writing file:', error);
    throw error;
  }
}

// READ - Load text from a file
export async function readFromFile(filename) {
  const filePath = getFilePath(filename);
  try {
    const content = await FileSystem.readAsStringAsync(filePath, {
      encoding: FileSystem.EncodingType.UTF8,
    });
    return content;
  } catch (error) {
    console.error('Error reading file:', error);
    return null;
  }
}

File Info and Existence Checks

// Check if a file exists
export async function fileExists(filename) {
  const filePath = getFilePath(filename);
  const info = await FileSystem.getInfoAsync(filePath);
  return info.exists;
}

// Get file information
export async function getFileInfo(filename) {
  const filePath = getFilePath(filename);
  const info = await FileSystem.getInfoAsync(filePath);

  if (info.exists) {
    console.log('File exists:', info.exists);
    console.log('Is directory:', info.isDirectory);
    console.log('Size (bytes):', info.size);
    console.log('Last modified:', info.modificationTime);
  }

  return info;
}

// List all files in document directory
export async function listFiles() {
  const files = await FileSystem.readDirectoryAsync(documentDirectory);
  console.log('Files:', files);
  return files;
}

Deleting and Moving Files

// Delete a file
export async function deleteFile(filename) {
  const filePath = getFilePath(filename);
  try {
    await FileSystem.deleteAsync(filePath);
    console.log('File deleted');
  } catch (error) {
    console.error('Error deleting file:', error);
  }
}

// Move a file
export async function moveFile(source, destination) {
  const sourcePath = getFilePath(source);
  const destPath = getFilePath(destination);

  try {
    await FileSystem.moveAsync({ from: sourcePath, to: destPath });
    console.log('File moved from', source, 'to', destination);
  } catch (error) {
    console.error('Error moving file:', error);
  }
}

// Copy a file
export async function copyFile(source, destination) {
  const sourcePath = getFilePath(source);
  const destPath = getFilePath(destination);

  try {
    await FileSystem.copyAsync({ from: sourcePath, to: destPath });
    console.log('File copied');
  } catch (error) {
    console.error('Error copying file:', error);
  }
}

// Create a directory
export async function createDirectory(dirname) {
  const dirPath = `${documentDirectory}${dirname}`;
  await FileSystem.makeDirectoryAsync(dirPath, { intermediates: true });
}

Downloading Files

downloadAsync downloads a file from a URL to a local path with progress tracking.

// Download a file
export async function downloadFile(url, filename) {
  const fileUri = `${documentDirectory}${filename}`;

  try {
    const { uri } = await FileSystem.downloadAsync(url, fileUri, {
      headers: {
        Authorization: 'Bearer your-token',
      },
    });
    console.log('Downloaded to:', uri);
    return uri;
  } catch (error) {
    console.error('Download error:', error);
    throw error;
  }
}

// Download with progress tracking
export async function downloadWithProgress(url, filename, onProgress) {
  const fileUri = `${documentDirectory}${filename}`;

  const downloadResumable = FileSystem.createDownloadResumable(
    url,
    fileUri,
    {
      headers: { Authorization: 'Bearer token' },
    },
    (downloadProgress) => {
      const progress = downloadProgress.totalBytesWritten / downloadProgress.totalBytesExpectedToWrite;
      onProgress(progress);
    }
  );

  try {
    const result = await downloadResumable.downloadAsync();
    return result.uri;
  } catch (error) {
    console.error('Download error:', error);
    throw error;
  }
}

// Pause and resume downloads
export async function pauseDownload(downloadResumable) {
  await downloadResumable.pauseAsync();
}

export async function resumeDownload(downloadResumable) {
  await downloadResumable.resumeAsync();
}

Using Downloads in Components

// screens/DownloadScreen.js
import { useState } from 'react';
import { View, Text, Button, ProgressViewIOS, Alert } from 'react-native';
import { downloadWithProgress } from '../utils/fileUtils';

export default function DownloadScreen() {
  const [progress, setProgress] = useState(0);
  const [downloading, setDownloading] = useState(false);

  const handleDownload = async () => {
    setDownloading(true);
    setProgress(0);

    try {
      await downloadWithProgress(
        'https://example.com/large-file.pdf',
        'document.pdf',
        (p) => setProgress(p)
      );
      Alert.alert('Success', 'File downloaded!');
    } catch (error) {
      Alert.alert('Error', 'Download failed');
    } finally {
      setDownloading(false);
    }
  };

  return (
    <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center' }}>
      <Text>Download Progress: {Math.round(progress * 100)}%</Text>
      <ProgressViewIOS progress={progress} />
      <Button
        title={downloading ? 'Downloading...' : 'Download File'}
        onPress={handleDownload}
        disabled={downloading}
      />
    </View>
  );
}

Summary

  • documentDirectory is for persistent files; cacheDirectory is temporary
  • Use writeAsStringAsync and readAsStringAsync for text files
  • getInfoAsync returns file metadata including size and existence
  • downloadAsync fetches files from URLs with progress tracking via createDownloadResumable
  • Use deleteAsync, moveAsync, and copyAsync for file management
  • Always handle errors when working with the file system
