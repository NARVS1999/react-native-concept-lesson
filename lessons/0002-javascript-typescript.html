JavaScript Fundamentals for React Native

Before diving into React Native, you need a solid grasp of core JavaScript concepts. Let's review the essentials.

Variables and Data Types

React Native uses let and const for variable declarations. Avoid var as it has function scope instead of block scope.

// const for values that won't change
const appName = 'MyApp';
const PI = 3.14159;

// let for values that will change
let counter = 0;
let userName = 'John';

// Data types
const isActive = true;          // boolean
const count = 42;               // number
const message = 'Hello!';      // string
const items = [1, 2, 3];       // array
const user = { name: 'John' }; // object
const nothing = null;           // null
const notDefined = undefined;   // undefined

Arrow Functions

Arrow functions are the standard way to write functions in React Native. They are shorter and lexically bind this.

// Traditional function
function add(a, b) {
  return a + b;
}

// Arrow function
const add = (a, b) => a + b;

// Arrow function with body
const greet = (name) => {
  const greeting = `Hello, ${name}!`;
  return greeting;
};

// Arrow function with single parameter (no parentheses needed)
const double = x => x * 2;

Array Methods

Array methods are essential for transforming data in React Native components.

const users = [
  { id: 1, name: 'Alice', age: 25 },
  { id: 2, name: 'Bob', age: 30 },
  { id: 3, name: 'Charlie', age: 35 },
];

// map - transform each item
const names = users.map(user => user.name);
// ['Alice', 'Bob', 'Charlie']

// filter - keep items that match condition
const adults = users.filter(user => user.age >= 30);
// [{ id: 2, name: 'Bob', age: 30 }, { id: 3, name: 'Charlie', age: 35 }]

// find - get first matching item
const bob = users.find(user => user.name === 'Bob');
// { id: 2, name: 'Bob', age: 30 }

// reduce - accumulate into single value
const totalAge = users.reduce((sum, user) => sum + user.age, 0);
// 90

Destructuring

Destructuring makes it easy to extract values from objects and arrays.

// Object destructuring
const user = { name: 'Alice', age: 25, email: 'alice@example.com' };
const { name, age, email } = user;

// With renaming
const { name: userName } = user;

// With defaults
const { role = 'user' } = user;

// Array destructuring
const colors = ['red', 'green', 'blue'];
const [first, second, third] = colors;

// In function parameters
const UserCard = ({ name, age }) => {
  return `${name} is ${age} years old`;
};

Spread and Rest Operators

// Spread operator - expand arrays/objects
const arr1 = [1, 2, 3];
const arr2 = [...arr1, 4, 5]; // [1, 2, 3, 4, 5]

const obj1 = { a: 1, b: 2 };
const obj2 = { ...obj1, c: 3 }; // { a: 1, b: 2, c: 3 }

// Rest operator - collect remaining items
const [first, ...rest] = [1, 2, 3, 4];
// first = 1, rest = [2, 3, 4]

const { id, ...userData } = { id: 1, name: 'Alice', age: 25 };
// id = 1, userData = { name: 'Alice', age: 25 }

TypeScript Basics

TypeScript is a superset of JavaScript that adds static type checking. React Native projects increasingly use TypeScript for better developer experience and fewer runtime errors.

Basic Types

// Primitive types
let name: string = 'Alice';
let age: number = 25;
let isActive: boolean = true;

// Array types
let scores: number[] = [90, 85, 92];
let names: Array<string> = ['Alice', 'Bob'];

// Union types
let id: string | number = 'abc123';
id = 42; // also valid

// Literal types
type Status = 'loading' | 'success' | 'error';
let currentStatus: Status = 'loading';

Interfaces

Interfaces define the shape of objects.

interface User {
  id: number;
  name: string;
  email: string;
  age?: number; // optional property
  role: 'admin' | 'user' | 'guest';
}

const user: User = {
  id: 1,
  name: 'Alice',
  email: 'alice@example.com',
  role: 'admin',
};

Generic Types

// Generic function
function getFirst<T>(arr: T[]): T | undefined {
  return arr[0];
}

const num = getFirst([1, 2, 3]); // number
const str = getFirst(['a', 'b']); // string

// Generic component-like pattern
interface ApiResponse<T> {
  data: T;
  status: number;
  message: string;
}

const userResponse: ApiResponse<User> = {
  data: { id: 1, name: 'Alice', email: 'a@b.com', role: 'user' },
  status: 200,
  message: 'Success',
};

JS vs TS Side-by-Side Comparison

// ============================================
// JavaScript Version
// ============================================
import React, { useState } from 'react';
import { View, Text, TextInput, Button } from 'react-native';

export default function LoginForm() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');

  const handleLogin = () => {
    // No type checking - could pass number, object, etc.
    console.log(email, password);
    // email could accidentally be set to a number
    setEmail(123); // No error in JS!
  };

  return (
    <View>
      <TextInput value={email} onChangeText={setEmail} />
      <TextInput value={password} onChangeText={setPassword} secureTextEntry />
      <Button title="Login" onPress={handleLogin} />
    </View>
  );
}

// ============================================
// TypeScript Version
// ============================================
import React, { useState } from 'react';
import { View, Text, TextInput, Button } from 'react-native';

interface LoginFormProps {
  onLogin: (email: string, password: string) => void;
}

export default function LoginForm({ onLogin }: LoginFormProps) {
  const [email, setEmail] = useState<string>('');
  const [password, setPassword] = useState<string>('');

  const handleLogin = () => {
    onLogin(email, password);
  };

  return (
    <View>
      <TextInput value={email} onChangeText={setEmail} />
      <TextInput value={password} onChangeText={setPassword} secureTextEntry />
      <Button title="Login" onPress={handleLogin} />
    </View>
  );
}

Why TypeScript is Preferred in React Native

  • Early Error Detection: Catch type-related bugs at compile time instead of runtime
  • Better IDE Support: Autocomplete, inline documentation, and refactoring tools work much better
  • Self-Documenting Code: Types serve as documentation for other developers (and your future self)
  • Refactoring Confidence: Change a prop name and TypeScript shows every place that needs updating
  • Industry Standard: Most modern React Native projects use TypeScript; knowing it makes you more employable

TS Configuration in Expo Projects

When you create a new Expo project, TypeScript is included by default. The configuration is in tsconfig.json:

{
  "extends": "expo/tsconfig.base",
  "compilerOptions": {
    "strict": true,
    "paths": {
      "@/*": ["./*"]
    }
  },
  "include": [
    "**/*.ts",
    "**/*.tsx",
    ".expo/types/**/*.ts",
    "expo-env.d.ts"
  ]
}

Key settings explained:

  • "strict": true — Enables all strict type-checking options (recommended)
  • "paths" — Allows import aliases like @/components/Button
  • "include" — Specifies which files TypeScript should check

Mini-Project: Convert a JS Component to TypeScript

Take this JavaScript component and convert it to TypeScript:

// Original: UserList.js (JavaScript)
import React from 'react';
import { View, Text, FlatList, StyleSheet } from 'react-native';

export default function UserList({ users, onSelect }) {
  return (
    <View style={styles.container}>
      <FlatList
        data={users}
        keyExtractor={(item) => item.id.toString()}
        renderItem={({ item }) => (
          <View style={styles.item}>
            <Text style={styles.name}>{item.name}</Text>
            <Text style={styles.email}>{item.email}</Text>
          </View>
        )}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1 },
  item: { padding: 15, borderBottomWidth: 1, borderBottomColor: '#eee' },
  name: { fontSize: 16, fontWeight: 'bold' },
  email: { fontSize: 14, color: '#666' },
});

Convert it to TypeScript by adding proper interfaces:

// Converted: UserList.tsx (TypeScript)
import React from 'react';
import { View, Text, FlatList, StyleSheet } from 'react-native';

interface User {
  id: number;
  name: string;
  email: string;
}

interface UserListProps {
  users: User[];
  onSelect: (user: User) => void;
}

export default function UserList({ users, onSelect }: UserListProps) {
  return (
    <View style={styles.container}>
      <FlatList
        data={users}
        keyExtractor={(item) => item.id.toString()}
        renderItem={({ item }) => (
          <View style={styles.item}>
            <Text style={styles.name}>{item.name}</Text>
            <Text style={styles.email}>{item.email}</Text>
          </View>
        )}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1 },
  item: { padding: 15, borderBottomWidth: 1, borderBottomColor: '#eee' },
  name: { fontSize: 16, fontWeight: 'bold' },
  email: { fontSize: 14, color: '#666' },
});

Summary

  • Use const and let instead of var for variable declarations
  • Arrow functions are the standard function syntax in React Native
  • Array methods like map, filter, and find are essential for data transformation
  • Destructuring makes extracting values from objects and arrays concise
  • TypeScript adds static type checking, catching bugs before they reach runtime
  • Interfaces define object shapes and are used extensively in React Native with TS
  • Expo projects include TypeScript support out of the box
  • TypeScript provides better IDE support, documentation, and refactoring capabilities
