Portrait vs Landscape Detection

When a user rotates their device, the screen dimensions change — width and height swap. React Native provides several ways to detect and respond to these orientation changes.

The simplest approach is comparing width and height: if width < height, the device is in portrait mode; if width > height, it's in landscape.

import { useWindowDimensions, View, Text, StyleSheet } from 'react-native';

function OrientationDetector() {
  const { width, height } = useWindowDimensions();
  const isPortrait = height > width;

  return (
    <View style={styles.container}>
      <Text style={styles.title}>
        {isPortrait ? '📱 Portrait' : '📲 Landscape'}
      </Text>
      <Text style={styles.dimension}>
        Width: {width} | Height: {height}
      </Text>
      <Text style={styles.dimension}>
        Orientation: {isPortrait ? 'Portrait' : 'Landscape'}
      </Text>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center', padding: 20 },
  title: { fontSize: 28, fontWeight: 'bold', marginBottom: 16 },
  dimension: { fontSize: 16, color: '#666', marginBottom: 8 },
});

The Dimensions API

The Dimensions API provides the screen's width and height. However, it's static — it doesn't update when the device rotates. Use it for initial measurements or one-time calculations.

import { Dimensions, Platform } from 'react-native';

// Get initial dimensions
const { width, height } = Dimensions.get('window');
const screenWidth = Dimensions.get('screen').width;
const screenHeight = Dimensions.get('screen').height;

console.log('Window:', width, height);
console.log('Screen:', screenWidth, screenHeight);
console.log('Platform:', Platform.OS);

// Listen for dimension changes (deprecated but still works)
const dimensionsHandler = ({ window }) => {
  console.log('New dimensions:', window.width, window.height);
};

Dimensions.addEventListener('change', dimensionsHandler);

// Better approach: use useWindowDimensions hook instead
function BetterDimensionHandler() {
  const { width, height } = useWindowDimensions(); // Automatically updates
  return null;
}

Using useWindowDimensions for Orientation

The useWindowDimensions hook is the recommended way to handle orientation changes. It automatically re-renders when the device rotates.

import { useState, useEffect } from 'react';
import { useWindowDimensions, View, Text, StyleSheet } from 'react-native';

function useOrientation() {
  const { width, height } = useWindowDimensions();
  const [orientation, setOrientation] = useState(
    width > height ? 'landscape' : 'portrait'
  );

  useEffect(() => {
    setOrientation(width > height ? 'landscape' : 'portrait');
  }, [width, height]);

  return {
    orientation,
    isPortrait: orientation === 'portrait',
    isLandscape: orientation === 'landscape',
    width,
    height,
  };
}

// Usage
function ResponsiveComponent() {
  const { isPortrait, isLandscape, width, height } = useOrientation();

  return (
    <View style={[styles.container, isLandscape && styles.landscape]}>
      <View style={[styles.box, isPortrait ? styles.boxVertical : styles.boxHorizontal]}>
        <Text style={styles.text}>
          {isPortrait ? 'Vertical Layout' : 'Horizontal Layout'}
        </Text>
      </View>
      <View style={[styles.sidebar, isPortrait && styles.sidebarHidden]}>
        <Text style={styles.text}>Sidebar</Text>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, flexDirection: 'column' },
  landscape: { flexDirection: 'row' },
  box: { padding: 20, backgroundColor: '#3b82f6' },
  boxVertical: { width: '100%' },
  boxHorizontal: { flex: 1 },
  sidebar: { width: 200, backgroundColor: '#1f2937', padding: 16 },
  sidebarHidden: { display: 'none' },
  text: { color: '#fff', fontSize: 16 },
});

Handling Orientation Changes

When the device rotates, you may need to adjust not just layout but also navigation behavior, keyboard handling, and scroll positions.

import { useEffect, useState } from 'react';
import { useWindowDimensions, Keyboard, Platform, StyleSheet } from 'react-native';

function useKeyboardAwareOrientation() {
  const { width, height } = useWindowDimensions();
  const [keyboardHeight, setKeyboardHeight] = useState(0);
  const isLandscape = width > height;

  useEffect(() => {
    if (Platform.OS === 'ios') {
      const showSub = Keyboard.addListener('keyboardWillShow', (e) => {
        setKeyboardHeight(e.endCoordinates.height);
      });
      const hideSub = Keyboard.addListener('keyboardWillHide', () => {
        setKeyboardHeight(0);
      });
      return () => {
        showSub.remove();
        hideSub.remove();
      };
    }
  }, []);

  return {
    isLandscape,
    isPortrait: !isLandscape,
    width,
    height,
    keyboardHeight,
    availableHeight: height - keyboardHeight,
  };
}

// Usage
function ChatScreen() {
  const { isLandscape, availableHeight } = useKeyboardAwareOrientation();

  return (
    <View style={[styles.container, { height: availableHeight }]}>
      {/* In landscape, show messages and input side by side */}
      <View style={[styles.messageArea, isLandscape && styles.messageAreaWide]}>
        {/* Messages list */}
      </View>
      <View style={[styles.inputArea, isLandscape && styles.inputAreaSide]}>
        {/* Input field */}
      </View>
    </View>
  );
}

Locking Orientation

Sometimes you want to lock the app to a specific orientation. This is configured in the app's native configuration, not in JavaScript.

// For Expo: app.json
{
  "expo": {
    "orientation": "portrait", // or "landscape" or "default"
    "ios": {
      "supportsTablet": true
    }
  }
}

// For React Native CLI (iOS): Info.plist
// Add UILaunchStoryboardName and set supported orientations

// Android: AndroidManifest.xml
// Add android:screenOrientation="portrait" to your activity

// Programmatic lock with expo-screen-orientation (Expo)
import * as ScreenOrientation from 'expo-screen-orientation';

async function lockToPortrait() {
  await ScreenOrientation.lockAsync(
    ScreenOrientation.OrientationLock.PORTRAIT_UP
  );
}

async function lockToLandscape() {
  await ScreenOrientation.lockAsync(
    ScreenOrientation.OrientationLock.LANDSCAPE_LEFT
  );
}

async function unlockOrientation() {
  await ScreenOrientation.unlockAsync();
}

Building an Orientation-Adaptive Component

Combine everything into a component that provides different experiences for portrait and landscape:

import { View, Text, Image, StyleSheet } from 'react-native';
import { useOrientation } from '../hooks/useOrientation';

function MediaCard({ title, description, imageUrl }) {
  const { isPortrait, isLandscape, width, height } = useOrientation();

  // Calculate image dimensions based on orientation
  const imageSize = isPortrait
    ? { width: width - 32, height: 200 }
    : { width: width * 0.4, height: height * 0.6 };

  return (
    <View style={[styles.card, isLandscape && styles.cardLandscape]}>
      <Image source={{ uri: imageUrl }} style={[styles.image, imageSize]} />
      <View style={[styles.content, isLandscape && styles.contentLandscape]}>
        <Text style={styles.title}>{title}</Text>
        <Text style={styles.description}>{description}</Text>
        {isLandscape && (
          <View style={styles.additionalInfo}>
            <Text style={styles.meta}>Additional info in landscape</Text>
          </View>
        )}
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  card: { margin: 16 },
  cardLandscape: { flexDirection: 'row', gap: 16 },
  image: { borderRadius: 12 },
  content: { marginTop: 12 },
  contentLandscape: { flex: 1, justifyContent: 'center' },
  title: { fontSize: 20, fontWeight: 'bold', marginBottom: 8 },
  description: { fontSize: 14, color: '#666' },
  additionalInfo: { marginTop: 12, padding: 12, backgroundColor: '#f3f4f6', borderRadius: 8 },
  meta: { fontSize: 12, color: '#999' },
});

Mini-Project: Build a Component That Adapts to Orientation

Create a component that changes its layout based on device orientation:

// hooks/useOrientation.js
import { useWindowDimensions } from 'react-native';

export default function useOrientation() {
  const { width, height } = useWindowDimensions();
  const isPortrait = height > width;

  return {
    isPortrait,
    isLandscape: !isPortrait,
    width,
    height,
  };
}

// components/OrientationCard.js
import { View, Text, Image, StyleSheet } from 'react-native';
import useOrientation from '../hooks/useOrientation';

export default function OrientationCard({ title, image }) {
  const { isPortrait } = useOrientation();

  return (
    <View style={[styles.card, isPortrait ? styles.vertical : styles.horizontal]}>
      <Image source={{ uri: image }} style={isPortrait ? styles.imageV : styles.imageH} />
      <View style={styles.content}>
        <Text style={styles.title}>{title}</Text>
        <Text style={styles.desc}>
          {isPortrait
            ? 'This card adapts to portrait mode with a vertical layout.'
            : 'This card adapts to landscape mode with a horizontal layout.'}
        </Text>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  card: { margin: 16, backgroundColor: '#fff', borderRadius: 12, overflow: 'hidden' },
  vertical: { alignItems: 'stretch' },
  horizontal: { flexDirection: 'row' },
  imageV: { width: '100%', height: 180 },
  imageH: { width: 140, height: 140 },
  content: { padding: 16 },
  title: { fontSize: 18, fontWeight: 'bold', marginBottom: 8 },
  desc: { fontSize: 14, color: '#666' },
});

Summary

  • Detect orientation by comparing width and height from useWindowDimensions
  • The Dimensions API is static and doesn't update on rotation — use useWindowDimensions instead
  • useWindowDimensions automatically re-renders components when the device rotates
  • Create a custom useOrientation hook for cleaner code throughout your app
  • Adjust layouts for portrait vs landscape (e.g., show sidebar only in landscape)
  • Lock orientation using native configuration or expo-screen-orientation library
