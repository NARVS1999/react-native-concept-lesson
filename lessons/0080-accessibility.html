Lesson 80: Accessibility in React Native

Phase 10 — Expert/Strategic Concepts

Lesson GoalLearn to make React Native apps accessible using built-in props, screen reader testing, and WCAG guidelines to ensure usability for all users.

Why Accessibility Matters

Accessibility ensures your app can be used by everyone, including people with disabilities. In React Native, you have powerful built-in tools to make your app accessible.

Key PrincipleAccessibility is not an afterthought—it should be integrated into your development process from the beginning.

// Basic accessibility
import { View, Text, TouchableOpacity, Image } from 'react-native';

function AccessibleComponent() {
  return (
    <View>
      {/* Make container accessible */}
      <View accessible={true}>
        <Text>This container is accessible</Text>
      </View>
      
      {/* TouchableOpacity is accessible by default */}
      <TouchableOpacity onPress={() => console.log('Pressed')}>
        <Text>Press me</Text>
      </TouchableOpacity>
      
      {/* Image needs alt text */}
      <Image
        source={{ uri: 'https://example.com/photo.jpg' }}
        accessibilityLabel="Photo of a sunset over the mountains"
        accessibilityRole="image"
      />
    </View>
  );
}

function NotificationBadge({ count, type }) {

  return (

// Complex state announcements
function FormField({ label, value, error, required }) {
  const accessibilityLabel = `${label}${required ? ' (required)' : ''}: ${value || 'empty'}`;
  const accessibilityHint = error ? `Error: ${error}` : undefined;
  
  return (
    <View>
      <Text accessibilityLabel={accessibilityLabel}>{label}</Text>
      <TextInput
        value={value}
        accessibilityLabel={accessibilityLabel}
        accessibilityHint={accessibilityHint}
        accessibilityState={{
          disabled: false,
          selected: false,
        }}
      />
      {error && (
        <Text
          accessibilityRole="alert"
          style={styles.errorText}
        >
          {error}
        </Text>
      )}
    </View>
  );
}

function AccessibleForm() {

        accessibilityRole="button"
        onPress={() => console.log('Pressed')}
      >
        <Text>Submit</Text>
      </TouchableOpacity>
      
      {/* Header roles */}
      <Text accessibilityRole="header">Main Title</Text>
      <Text accessibilityRole="header">Subtitle</Text>
      
      {/* Link role */}
      <Text
        accessibilityRole="link"
        onPress={() => Linking.openURL('https://example.com')}
      >
        Visit our website
      </Text>
      
      {/* Image role */}
      <Image
        source={{ uri: 'logo.png' }}
        accessibilityRole="image"
        accessibilityLabel="Company logo"
      />
      
      {/* Switch role */}
      <Switch
        value={isEnabled}
        onValueChange={setIsEnabled}
        accessibilityLabel="Enable notifications"
        accessibilityRole="switch"
      />
      
      {/* Search role */}
      <TextInput
        accessibilityRole="search"
        placeholder="Search..."
      />
      
      {/* Summary role */}
      <Text accessibilityRole="summary">
        3 items in your cart totaling $45.99
      </Text>
      
      {/* Header and content grouping */}

        <Text accessibilityRole="header">Section Title</Text>

        <View accessibilityRole="listitem">
          <Text>Item 1</Text>

  return (

      accessibilityHint="Double tap to submit the form"
      accessibilityState={{
        disabled: isLoading,
        selected: isSelected,
        checked: undefined,
        busy: isLoading,
        expanded: undefined,
      }}
      onPress={handleSubmit}
    >
      {isLoading ? (
        <ActivityIndicator accessibilityLabel="Loading" />
      ) : (
        <Text>Submit</Text>
      )}
    </TouchableOpacity>
  );
}

// Checkbox state
function Checkbox({ label, checked, onChange }) {
  return (
    <TouchableOpacity
      accessibilityRole="checkbox"
      accessibilityLabel={label}
      accessibilityState={{
        checked: checked,
      }}
      onPress={() => onChange(!checked)}
    >
      <View style={styles.checkbox}>
        {checked && <Text>✓</Text>}
      </View>
      <Text>{label}</Text>
    </TouchableOpacity>
  );
}

// Expandable section
function ExpandableSection({ title, content }) {
  const [isExpanded, setIsExpanded] = useState(false);
  
  return (
    <View>
      <TouchableOpacity
        accessibilityLabel={title}
        accessibilityHint={isExpanded ? "Collapse section" : "Expand section"}
        accessibilityState={{
          expanded: isExpanded,
        }}
        onPress={() => setIsExpanded(!isExpanded)}
      >
        <Text>{title}</Text>
        <Text>{isExpanded ? '▼' : '▶'}</Text>
      </TouchableOpacity>
      
      {isExpanded && (
        <View accessibilityLabel={`${title} content`}>
          <Text>{content}</Text>
        </View>
      )}
    </View>
  );
}

WCAG Guidelines for Mobile

  • Color Contrast: Ensure 4.5:1 contrast ratio for normal text
  • Touch Targets: Minimum 44x44 points for interactive elements
  • Text Scaling: Support dynamic type/text sizes
  • Focus Management: Logical navigation order
  • Error Identification: Clear error messages and recovery
  • Consistent Navigation: Predictable interface behavior

// components/AccessibleTextInput.js
import React from 'react';
import { View, Text, TextInput, StyleSheet } from 'react-native';

export function AccessibleTextInput({
  label,
  value,
  onChangeText,
  error,
  required,
  secureTextEntry,
  keyboardType,
  autoCapitalize,
  autoComplete,
  accessibilityHint,
}) {
  const fieldId = label.toLowerCase().replace(/\s+/g, '_');
  
  return (
    <View style={styles.fieldContainer}>
      <Text
        accessibilityLabel={`${label}${required ? ' (required)' : ''}`}
        style={styles.label}
      >
        {label}
        {required && <Text style={styles.required}> *</Text>}
      </Text>
      
      <TextInput
        value={value}
        onChangeText={onChangeText}
        accessibilityLabel={`${label} input`}
        accessibilityHint={accessibilityHint || `Enter your ${label.toLowerCase()}`}
        accessibilityRequired={required}
        secureTextEntry={secureTextEntry}
        keyboardType={keyboardType}
        autoCapitalize={autoCapitalize}
        autoComplete={autoComplete}
        accessibilityState={{
          disabled: false,
        }}
        style={[
          styles.input,
          error && styles.inputError,
        ]}
      />
      
      {error && (
        <Text
          accessibilityRole="alert"
          accessibilityLabel={`${label} error: ${error}`}
          style={styles.errorText}
        >
          {error}
        </Text>
      )}
    </View>
  );
}

        checked: checked,
      }}

    </TouchableOpacity>
  );
}

// screens/RegistrationScreen.js
export function RegistrationScreen() {
  const [formData, setFormData] = useState({
    name: '',
    email: '',
    password: '',
    confirmPassword: '',
    acceptTerms: false,
  });
  const [errors, setErrors] = useState({});
  const [isSubmitting, setIsSubmitting] = useState(false);
  
  const validateForm = () => {
    const newErrors = {};
    
    if (!formData.name.trim()) {
      newErrors.name = 'Name is required';
    }
    
    if (!formData.email.trim()) {
      newErrors.email = 'Email is required';
    } else if (!/\S+@\S+\.\S+/.test(formData.email)) {
      newErrors.email = 'Please enter a valid email address';
    }
    
    if (!formData.password) {
      newErrors.password = 'Password is required';
    } else if (formData.password.length < 8) {
      newErrors.password = 'Password must be at least 8 characters';
    }
    
    if (formData.password !== formData.confirmPassword) {
      newErrors.confirmPassword = 'Passwords do not match';
    }
    
    if (!formData.acceptTerms) {
      newErrors.acceptTerms = 'You must accept the terms and conditions';
    }
    
    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };
  
  const handleSubmit = async () => {
    if (!validateForm()) {
      // Announce error summary for screen readers
      const errorCount = Object.keys(errors).length;
      Alert.alert(
        'Form Errors',
        `Please correct ${errorCount} error${errorCount > 1 ? 's' : ''} in the form.`
      );
      return;
    }
    
    setIsSubmitting(true);
    try {
      await registerUser(formData);
      Alert.alert('Success', 'Registration complete!');
    } catch (error) {
      setErrors({ general: error.message });
    } finally {
      setIsSubmitting(false);
    }
  };
  
  return (
    <ScrollView
      contentContainerStyle={styles.container}
      keyboardShouldPersistTaps="handled"
    >
      <Text accessibilityRole="header" style={styles.title}>
        Create Account
      </Text>
      
      {errors.general && (
        <View
          accessibilityRole="alert"
          style={styles.generalError}
        >
          <Text style={styles.generalErrorText}>
            {errors.general}
          </Text>
        </View>
      )}
      
      <AccessibleTextInput
        label="Full Name"
        value={formData.name}
        onChangeText={(value) => setFormData({...formData, name: value})}
        error={errors.name}
        required
        autoCapitalize="words"
        autoComplete="name"
        accessibilityHint="Enter your full name as it appears on your ID"
      />
      
      <AccessibleTextInput
        label="Email Address"
        value={formData.email}
        onChangeText={(value) => setFormData({...formData, email: value})}
        error={errors.email}
        required
        keyboardType="email-address"
        autoCapitalize="none"
        autoComplete="email"
        accessibilityHint="Enter a valid email address for account verification"
      />
      
      <AccessibleTextInput
        label="Password"
        value={formData.password}
        onChangeText={(value) => setFormData({...formData, password: value})}
        error={errors.password}
        required
        secureTextEntry
        autoCapitalize="none"
        autoComplete="password"
        accessibilityHint="Enter a password with at least 8 characters"
      />
      
      <AccessibleTextInput
        label="Confirm Password"
        value={formData.confirmPassword}
        onChangeText={(value) => setFormData({...formData, confirmPassword: value})}
        error={errors.confirmPassword}
        required
        secureTextEntry
        autoCapitalize="none"
        autoComplete="password"
        accessibilityHint="Re-enter your password to confirm"
      />
      
      <AccessibleCheckbox
        label="I agree to the Terms of Service and Privacy Policy"
        checked={formData.acceptTerms}
        onChange={(value) => setFormData({...formData, acceptTerms: value})}
        error={errors.acceptTerms}
        required
      />
      
      <TouchableOpacity
        accessibilityRole="button"
        accessibilityLabel="Create account"
        accessibilityHint="Double tap to create your account"
        accessibilityState={{
          disabled: isSubmitting,
          busy: isSubmitting,
        }}
        onPress={handleSubmit}
        style={[styles.button, isSubmitting && styles.buttonDisabled]}
        disabled={isSubmitting}
      >
        {isSubmitting ? (
          <ActivityIndicator color="#fff" accessibilityLabel="Creating account" />
        ) : (
          <Text style={styles.buttonText}>Create Account</Text>
        )}
      </TouchableOpacity>
      
      <TouchableOpacity
        accessibilityRole="link"
        accessibilityLabel="Already have an account? Sign in"
        accessibilityHint="Navigates to login screen"
        onPress={() => navigation.navigate('Login')}
        style={styles.linkButton}
      >
        <Text style={styles.linkText}>
          Already have an account? Sign in
        </Text>
      </TouchableOpacity>
    </ScrollView>
  );
}

Summary

  • Use accessibilityLabel to provide descriptive text for screen readers
  • Apply appropriate accessibilityRole for semantic meaning
  • Use accessibilityHint to provide additional context for actions
  • Communicate component state with accessibilityState
  • Follow WCAG guidelines: 4.5:1 contrast ratio, 44x44pt touch targets
  • Test with VoiceOver (iOS) and TalkBack (Android)
  • Ensure logical focus order and keyboard navigation
  • Provide clear error identification and recovery instructions
