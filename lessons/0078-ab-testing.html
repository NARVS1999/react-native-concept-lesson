What is A/B Testing?

A/B testing compares two versions of a feature to determine which performs better. In React Native, you can implement A/B testing using feature flags to show different variants to different user segments.

Key ConceptA/B testing is not just about UI changes—it's about measuring business outcomes like conversion rates, engagement, and retention.

// Assignment logic
// utils/experimentAssigner.js
import AsyncStorage from '@react-native-async-storage/async-storage';

export async function assignVariant(experimentId, userId) {
  const experiment = experiments[experimentId];
  if (!experiment) return null;

  // Check targeting rules
  if (!isTargeted(experiment.targeting, userId)) {
    return null;
  }

  // Check for existing assignment
  const existing = await AsyncStorage.getItem(`experiment_${experimentId}`);
  if (existing) return JSON.parse(existing);

  // Assign variant based on weights
  const variant = weightedRandom(experiment.variants);
  
  // Store assignment
  await AsyncStorage.setItem(
    `experiment_${experimentId}`,
    JSON.stringify({ variant, assignedAt: Date.now() })
  );

  // Track assignment event
  trackExperimentAssignment(experimentId, variant);
  
  return variant;
}

function weightedRandom(variants) {
  const totalWeight = Object.values(variants).reduce(
    (sum, v) => sum + v.weight, 0
  );
  let random = Math.random() * totalWeight;
  
  for (const [name, variant] of Object.entries(variants)) {
    random -= variant.weight;
    if (random <= 0) return name;
  }
  
  return Object.keys(variants)[0];
}

// React hook for experiments
// hooks/useExperiment.js
import { useState, useEffect } from 'react';
import { assignVariant } from '../utils/experimentAssigner';

export function useExperiment(experimentId, userId) {
  const [variant, setVariant] = useState(null);
  const [config, setConfig] = useState(null);

  useEffect(() => {
    async function loadExperiment() {
      const assignedVariant = await assignVariant(experimentId, userId);
      if (assignedVariant) {
        setVariant(assignedVariant);
        setConfig(experiments[experimentId].variants[assignedVariant].config);
      }
    }
    loadExperiment();
  }, [experimentId, userId]);

  return { variant, config };
}

// Usage in component
function CheckoutScreen({ userId }) {

  return (
    <View>
      {config.showExpressCheckout && <ExpressCheckoutButton />}
      <CheckoutLayout type={config.checkoutLayout} />
    </View>
  );
}

Firebase Remote Config

Firebase Remote Config provides a powerful way to manage A/B tests with server-side control:

// Firebase Remote Config setup
// config/firebase.js
import firebase from '@react-native-firebase/app';
import remoteConfig from '@react-native-firebase/remote-config';

// Initialize Firebase
if (!firebase.apps.length) {
  firebase.initializeApp({
    // Your Firebase config
  });
}

// Fetch and activate config
async function initializeRemoteConfig() {
  await remoteConfig().setConfigSettings({
    minimumFetchInterval: 3600, // 1 hour
  });

  await remoteConfig().setDefaults({
    checkout_flow_enabled: false,
    checkout_flow_variant: 'control',
    new_ui_enabled: false,
  });

  try {
    await remoteConfig().fetchAndActivate();
  } catch (error) {
    console.error('Remote config fetch failed:', error);
  }
}

// Get experiment variant from Remote Config
async function getExperimentVariant(experimentId) {
  const enabled = remoteConfig().getBool(`${experimentId}_enabled`);
  if (!enabled) return 'control';
  
  return remoteConfig().getString(`${experimentId}_variant`);
}

// Usage
async function CheckoutScreen() {
  const variant = await getExperimentVariant('checkout_flow');
  
  return (
    <View>
      {variant === 'treatment' ? (
        <NewCheckoutFlow />
      ) : (
        <OldCheckoutFlow />
      )}
    </View>
  );
}

// Listen for config changes
remoteConfig().onConfigUpdated(async (result) => {
  if (result.error) {
    console.error('Config update error:', result.error);
    return;
  }
  
  await remoteConfig().activate();
  
  // Re-render components with new config
  updateExperimentUI();
});

Measuring Experiment Results

// Analytics tracking for experiments
// analytics/experimentTracker.js
import analytics from '@react-native-firebase/analytics';

class ExperimentTracker {
  static trackAssignment(experimentId, variant, userId) {
    analytics().logEvent('experiment_assigned', {
      experiment_id: experimentId,
      variant: variant,
      user_id: userId,
      timestamp: Date.now(),
    });
  }

  static trackConversion(experimentId, variant, goal, value = 1) {
    analytics().logEvent('experiment_conversion', {
      experiment_id: experimentId,
      variant: variant,
      goal: goal,
      value: value,
      timestamp: Date.now(),
    });
  }

  static trackMetric(experimentId, variant, metricName, metricValue) {
    analytics().logEvent('experiment_metric', {
      experiment_id: experimentId,
      variant: variant,
      metric_name: metricName,
      metric_value: metricValue,
      timestamp: Date.now(),
    });
  }
}

// Statistical significance calculator
function calculateSignificance(control, treatment) {
  const n1 = control.sampleSize;
  const n2 = treatment.sampleSize;
  const p1 = control.conversionRate;
  const p2 = treatment.conversionRate;
  
  const pooledP = (p1 * n1 + p2 * n2) / (n1 + n2);
  const standardError = Math.sqrt(
    pooledP * (1 - pooledP) * (1/n1 + 1/n2)
  );
  
  const zScore = (p2 - p1) / standardError;
  const pValue = 2 * (1 - normalCDF(Math.abs(zScore)));
  
  return {
    significant: pValue < 0.05,
    confidence: (1 - pValue) * 100,
    lift: ((p2 - p1) / p1) * 100,
    pValue,
  };
}

// Experiment analysis hook
function useExperimentAnalysis(experimentId) {
  const [results, setResults] = useState(null);
  
  useEffect(() => {
    async function analyzeExperiment() {
      const data = await fetchExperimentData(experimentId);
      const analysis = {
        control: calculateMetrics(data.control),
        treatment: calculateMetrics(data.treatment),
      };
      
      analysis.significance = calculateSignificance(
        analysis.control,
        analysis.treatment
      );
      
      setResults(analysis);
    }
    
    analyzeExperiment();
  }, [experimentId]);
  
  return results;
}

Implement a complete A/B testing system for a button color experiment:

// config/buttonExperiment.js
export const BUTTON_EXPERIMENT = {
  id: 'button_color_test',
  variants: {
    control: {
      weight: 50,
      color: '#007AFF', // iOS blue
    },
    treatment_a: {
      weight: 25,
      color: '#34C759', // iOS green
    },
    treatment_b: {
      weight: 25,
      color: '#FF9500', // iOS orange
    },
  },
};

// 2. Create experiment hook
// hooks/useButtonExperiment.js
import { useState, useEffect } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';

export function useButtonExperiment(userId) {
  const [variant, setVariant] = useState(null);

  useEffect(() => {
    async function assignVariant() {
      const key = `button_experiment_${userId}`;
      const existing = await AsyncStorage.getItem(key);
      
      if (existing) {
        setVariant(JSON.parse(existing));
        return;
      }
      
      // Assign based on weights
      const totalWeight = 100;
      const random = Math.random() * totalWeight;
      
      let assigned;
      if (random < 50) assigned = 'control';
      else if (random < 75) assigned = 'treatment_a';
      else assigned = 'treatment_b';
      
      await AsyncStorage.setItem(key, JSON.stringify(assigned));
      setVariant(assigned);
    }
    
    assignVariant();
  }, [userId]);

  return variant;
}
