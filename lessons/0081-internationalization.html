Lesson 81: Internationalization (i18n)

Phase 10 — Expert/Strategic Concepts

Lesson GoalLearn to implement internationalization in React Native using i18next, translation files, pluralization, and RTL support for global audiences.

Why Internationalization?

Internationalization (i18n) prepares your app for localization—adapting it to different languages, regions, and cultures. This includes translating text, formatting dates/numbers, and supporting different text directions.

Key DifferenceInternationalization is the technical preparation; localization is the actual translation and cultural adaptation.

// Language detector
const languageDetector = {
  type: 'languageDetector',
  async: true,
  detect: async (callback) => {
    try {
      const savedLanguage = await AsyncStorage.getItem('userLanguage');
      if (savedLanguage) {
        callback(savedLanguage);
        return;
      }
      
      // Use device language
      const deviceLanguage = NativeModules.I18nManager?.localeIdentifier || 'en';
      callback(deviceLanguage.split('-')[0]);
    } catch (error) {
      callback('en');
    }
  },
  init: () => {},
  cacheUserLanguage: async (language) => {
    try {
      await AsyncStorage.setItem('userLanguage', language);
    } catch (error) {
      console.error('Error saving language:', error);
    }
  },
};

// Time display with pluralization
function TimeAgo({ timestamp }) {
  const { t } = useTranslation();
  
  const getRelativeTime = (timestamp) => {
    const now = Date.now();
    const diff = now - timestamp;
    
    const minutes = Math.floor(diff / 60000);
    const hours = Math.floor(diff / 3600000);
    const days = Math.floor(diff / 86400000);
    
    if (minutes < 1) return t('minutes', { count: 0 });
    if (minutes < 60) return t('minutes', { count: minutes });
    if (hours < 24) return t('hours', { count: hours });
    return t('days', { count: days });
  };
  
  return (
    <Text>{getRelativeTime(timestamp)}</Text>
  );
}

// Custom date formatter
function formatLocalDate(date, locale, format = 'medium') {
  const options = {
    short: { month: 'numeric', day: 'numeric' },
    medium: { year: 'numeric', month: 'short', day: 'numeric' },
    long: { year: 'numeric', month: 'long', day: 'numeric', weekday: 'long' },
    full: {
      year: 'numeric',
      month: 'long',
      day: 'numeric',
      weekday: 'long',
      hour: '2-digit',
      minute: '2-digit',
    },
  };
  
  return new Intl.DateTimeFormat(locale, options[format]).format(date);
}

// Number formatting
function formatLocalNumber(number, locale, style = 'decimal') {
  const options = {
    decimal: { style: 'decimal' },
    currency: { style: 'currency', currency: getCurrencyForLocale(locale) },
    percent: { style: 'percent' },
    compact: { notation: 'compact', compactDisplay: 'short' },
  };
  
  return new Intl.NumberFormat(locale, options[style]).format(number);
}

// Currency mapping
function getCurrencyForLocale(locale) {
  const currencies = {
    en: 'USD',
    es: 'EUR',
    fr: 'EUR',
    ja: 'JPY',
  };
  return currencies[locale] || 'USD';
}

// Usage in components
function PriceDisplay({ amount, locale }) {
  const formattedPrice = formatLocalNumber(amount, locale, 'currency');
  
  return (
    <Text style={styles.price}>{formattedPrice}</Text>
  );
}

function DateFormat({ date, locale }) {
  const formattedDate = formatLocalDate(date, locale, 'long');
  
  return (
    <Text>{formattedDate}</Text>
  );
}

// i18n/formatters.js
i18n.services.formatter.add('datetime', (value, lng, options) => {

    year: 'numeric',
    month: 'long',
    day: 'numeric',

RTL (Right-to-Left) Support

// RTL layout support
import { I18nManager, Platform } from 'react-native';

// Configure RTL
function configureRTL(language) {
  const rtlLanguages = ['ar', 'he', 'fa', 'ur'];
  const isRTL = rtlLanguages.includes(language);
  
  if (I18nManager.isRTL !== isRTL) {
    I18nManager.forceRTL(isRTL);
    I18nManager.allowRTL(isRTL);
    
    // Restart app for changes to take effect
    if (Platform.OS === 'ios') {
      // iOS requires app restart
      Alert.alert(
        'Language Change',
        'Please restart the app to apply language changes.',
        [{ text: 'OK', onPress: () => RNRestart.restart() }]
      );
    } else {
      // Android can sometimes apply without restart
      RNRestart.restart();
    }
  }
}

// Usage in language change handler
async function changeLanguage(language) {
  await i18n.changeLanguage(language);
  configureRTL(language);
}

// Styled components with RTL support
import { StyleSheet } from 'react-native';

const styles = StyleSheet.create({
  container: {
    flexDirection: I18nManager.isRTL ? 'row-reverse' : 'row',
    alignItems: 'center',
  },
  text: {
    textAlign: I18nManager.isRTL ? 'right' : 'left',
  },
  icon: {
    marginStart: I18nManager.isRTL ? 0 : 10,
    marginEnd: I18nManager.isRTL ? 10 : 0,
  },
});

// Custom hook for RTL
function useRTL() {
  const { i18n } = useTranslation();
  const isRTL = I18nManager.isRTL;
  
  return {
    isRTL,
    direction: isRTL ? 'rtl' : 'ltr',
    textAlign: isRTL ? 'right' : 'left',
    flexDirection: isRTL ? 'row-reverse' : 'row',
  };
}

// Usage in component
function RTLExample() {
  const { isRTL, textAlign, flexDirection } = useRTL();
  
  return (
    <View style={{ flexDirection }}>
      <Text style={{ textAlign }}>
        {isRTL ? 'العربية' : 'English'}
      </Text>
      <Icon name="arrow-forward" />
    </View>
  );
}

Language Detection and Switching

// Complete language management
// services/languageService.js
import i18n from '../i18n';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { I18nManager, NativeModules } from 'react-native';

class LanguageService {
  static supportedLanguages = [
    { code: 'en', name: 'English', nativeName: 'English', rtl: false },
    { code: 'es', name: 'Spanish', nativeName: 'Español', rtl: false },
    { code: 'fr', name: 'French', nativeName: 'Français', rtl: false },
    { code: 'ja', name: 'Japanese', nativeName: '日本語', rtl: false },
    { code: 'ar', name: 'Arabic', nativeName: 'العربية', rtl: true },
  ];

  static async initialize() {
    // Check for saved language
    const savedLanguage = await AsyncStorage.getItem('appLanguage');
    if (savedLanguage) {
      await this.setLanguage(savedLanguage);
      return;
    }

    // Detect device language
    const deviceLanguage = this.getDeviceLanguage();
    const supportedLanguage = this.findSupportedLanguage(deviceLanguage);
    
    if (supportedLanguage) {
      await this.setLanguage(supportedLanguage.code);
    } else {
      await this.setLanguage('en');
    }
  }

  static getDeviceLanguage() {
    if (Platform.OS === 'ios') {
      return NativeModules.SettingsManager?.settings?.AppleLocale?.split('-')[0] || 'en';
    }
    return NativeModules.I18nManager?.localeIdentifier?.split('-')[0] || 'en';
  }

  static findSupportedLanguage(languageCode) {
    return this.supportedLanguages.find(lang => lang.code === languageCode);
  }

  static async setLanguage(languageCode) {
    const language = this.findSupportedLanguage(languageCode);
    if (!language) return false;

    await i18n.changeLanguage(languageCode);
    await AsyncStorage.setItem('appLanguage', languageCode);
    
    // Handle RTL
    if (I18nManager.isRTL !== language.rtl) {
      I18nManager.forceRTL(language.rtl);
      I18nManager.allowRTL(language.rtl);
    }

    return true;
  }

  static getCurrentLanguage() {
    const code = i18n.language;
    return this.findSupportedLanguage(code) || this.supportedLanguages[0];
  }

  static isRTL() {
    return I18nManager.isRTL;
  }
}

// Language context
// contexts/LanguageContext.js
import React, { createContext, useContext, useState, useEffect } from 'react';
import LanguageService from '../services/languageService';

const LanguageContext = createContext();

export function LanguageProvider({ children }) {
  const [currentLanguage, setCurrentLanguage] = useState(
    LanguageService.getCurrentLanguage()
  );

  useEffect(() => {
    LanguageService.initialize();
  }, []);

  const changeLanguage = async (languageCode) => {
    const success = await LanguageService.setLanguage(languageCode);
    if (success) {
      setCurrentLanguage(LanguageService.getCurrentLanguage());
    }
    return success;
  };

  return (
    <LanguageContext.Provider
      value={{
        currentLanguage,
        changeLanguage,
        supportedLanguages: LanguageService.supportedLanguages,
        isRTL: LanguageService.isRTL(),
      }}
    >
      {children}
    </LanguageContext.Provider>
  );
}

export function useLanguage() {
  const context = useContext(LanguageContext);
  if (!context) {
    throw new Error('useLanguage must be used within a LanguageProvider');
  }
  return context;
}

// Usage in App.js
function App() {
  return (
    <LanguageProvider>
      <NavigationContainer>
        {/* Your app navigation */}
      </NavigationContainer>
    </LanguageProvider>
  );
}

// 3. Create language context
// contexts/LanguageContext.js
import React, { createContext, useContext, useState, useEffect } from 'react';
import i18n from '../i18n';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { I18nManager } from 'react-native';

const LanguageContext = createContext();

export function LanguageProvider({ children }) {
  const [language, setLanguage] = useState(i18n.language);
  const [isRTL, setIsRTL] = useState(false);

  useEffect(() => {
    loadSavedLanguage();
  }, []);

  const loadSavedLanguage = async () => {
    try {
      const saved = await AsyncStorage.getItem('appLanguage');
      if (saved) {
        await changeLanguage(saved);
      }
    } catch (error) {
      console.error('Error loading language:', error);
    }
  };

  const changeLanguage = async (langCode) => {
    await i18n.changeLanguage(langCode);
    await AsyncStorage.setItem('appLanguage', langCode);
    setLanguage(langCode);
    
    // Handle RTL
    const rtlLanguages = ['ar', 'he', 'fa'];
    const shouldBeRTL = rtlLanguages.includes(langCode);
    
    if (I18nManager.isRTL !== shouldBeRTL) {
      I18nManager.forceRTL(shouldBeRTL);
      setIsRTL(shouldBeRTL);
    }
  };

  return (
    <LanguageContext.Provider
      value={{
        language,
        changeLanguage,
        isRTL,
        t: i18n.t.bind(i18n),
      }}
    >
      {children}
    </LanguageContext.Provider>
  );
}

export function useLanguage() {
  return useContext(LanguageContext);
}

Summary

  • i18next provides a complete internationalization solution for React Native
  • Use translation files to organize and manage translations
  • Pluralization handles singular/plural forms across languages
  • Date and number formatting should respect locale conventions
  • RTL support requires I18nManager configuration and app restart
  • Language detection can use device settings or user preferences
  • Always provide fallback languages for missing translations
  • Test with native speakers for accurate translations
