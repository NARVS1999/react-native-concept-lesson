Analytics Providers Overview

Choosing the right analytics provider depends on your needs. Here's a comparison of popular options:

  • Firebase Analytics — Free, Google-integrated, good for basic analytics
  • Mixpanel — Advanced user behavior tracking, great for product analytics
  • Amplitude — Strong cohort analysis and retention metrics
  • Segment — Data routing to multiple analytics tools

  });

// Example: Track purchase
export function logPurchase(transactionId, items, total) {
  analytics().logPurchase({
    transaction_id: transactionId,
    value: total,
    currency: 'USD',
    items: items.map(item => ({
      item_id: item.id,
      item_name: item.name,
      price: item.price,
      quantity: item.quantity,
    })),
  });
}

// Set user properties

  Object.entries(properties).forEach(([key, value]) => {
    analytics().setUserProperty(key, String(value));
  });
}

Mixpanel Integration

// Mixpanel setup
// analytics/mixpanel.js
import Mixpanel from 'react-native-mixpanel';

// Initialize Mixpanel
export function initializeMixpanel(token) {
  Mixpanel.optOutTracking(); // Default to opted out
  Mixpanel.initialize(token, {
    trackAutomaticEvents: true,
    useNative: true,
  });
}

// Track events
export function track(event, properties = {}) {
  Mixpanel.trackWithProperties(event, {
    ...properties,
    timestamp: new Date().toISOString(),
  });
}

// Identify user
export function identify(userId, properties = {}) {
  Mixpanel.identify(userId);
  Mixpanel.set(properties);
}

// Track funnel steps
export function trackFunnelStep(funnelName, step, properties = {}) {
  track('Funnel Step', {
    funnel_name: funnelName,
    step: step,
    ...properties,
  });
}

// Track revenue
export function trackRevenue(amount, productId) {
  Mixpanel.trackWithProperties('Purchase', {
    $amount: amount,
    product_id: productId,
  });
}

// Example: Onboarding funnel
function OnboardingScreen() {
  const handleStepComplete = (step) => {
    trackFunnelStep('onboarding', step, {
      user_type: 'new',
    });
  };

  return (
    <View>
      {/* Onboarding steps */}
      <TouchableOpacity onPress={() => handleStepComplete('step_1')}>
        <Text>Complete Step 1</Text>
      </TouchableOpacity>
    </View>
  );
}

User Properties and Cohorts

// User property management
// analytics/userProperties.js
class UserPropertyManager {
  static properties = {};
  
  static set(key, value) {
    this.properties[key] = value;
    this.syncToProviders(key, value);
  }
  
  static get(key) {
    return this.properties[key];
  }
  
  static syncToProviders(key, value) {
    // Firebase
    analytics().setUserProperty(key, String(value));
    
    // Mixpanel
    Mixpanel.set({ [key]: value });
    
    // Amplitude
    AmplitudeReactNative.setUserProperties({ [key]: value });
  }
  
  static identify(userId) {
    // Set user ID across all providers
    analytics().setUserId(userId);
    Mixpanel.identify(userId);
    AmplitudeReactNative.setUserId(userId);
  }
}

// Usage
UserPropertyManager.identify('user_123');
UserPropertyManager.set('subscription_plan', 'premium');
UserPropertyManager.set('company_size', 'enterprise');
UserPropertyManager.set('industry', 'technology');

// Cohort-based tracking
function trackCohort(cohortName, properties = {}) {

}

Funnels and Retention

// Funnel tracking
// analytics/funnels.js
class FunnelTracker {
  static funnels = {};
  
  static startFunnel(funnelName, userId) {
    this.funnels[funnelName] = {
      userId,
      startTime: Date.now(),
      steps: [],
    };
  }
  
  static trackStep(funnelName, stepName, properties = {}) {
    const funnel = this.funnels[funnelName];
    if (!funnel) return;
    
    funnel.steps.push({
      step: stepName,
      timestamp: Date.now(),
      ...properties,
    });
    
    track('Funnel Step', {
      funnel_name: funnelName,
      step: stepName,
      step_number: funnel.steps.length,
      ...properties,
    });
  }
  
  static completeFunnel(funnelName, success = true) {
    const funnel = this.funnels[funnelName];
    if (!funnel) return;
    
    const duration = Date.now() - funnel.startTime;
    
    track('Funnel Completed', {
      funnel_name: funnelName,
      completed: success,
      duration_ms: duration,
      steps_completed: funnel.steps.length,
    });
    
    delete this.funnels[funnelName];
  }
}

// Retention tracking
function trackRetention(userId, daysSinceInstall) {
  track('Daily Active', {
    user_id: userId,
    days_since_install: daysSinceInstall,
    retention_day: daysSinceInstall,
  });
}

// Example: E-commerce funnel
function CheckoutFlow() {
  useEffect(() => {
    FunnelTracker.startFunnel('checkout', userId);
  }, []);
  
  const handleShippingSubmit = () => {
    FunnelTracker.trackStep('checkout', 'shipping_info', {
      shipping_method: 'standard',
    });
    // Proceed to payment
  };
  
  const handlePaymentSubmit = () => {
    FunnelTracker.trackStep('checkout', 'payment_info');
    // Process payment
  };
  
  const handleOrderComplete = () => {
    FunnelTracker.completeFunnel('checkout', true);
  };
  
  return (
    <View>
      {/* Checkout steps */}
    </View>
  );
}

Privacy Considerations

// Privacy-first analytics
// analytics/privacy.js
class PrivacyManager {
  static hasConsent = false;
  
  static async requestConsent() {
    // Show consent dialog
    const consent = await showConsentDialog();
    this.hasConsent = consent;
    
    if (consent) {
      this.enableTracking();
    } else {
      this.disableTracking();
    }
    
    return consent;
  }
  
  static enableTracking() {
    analytics().setAnalyticsCollectionEnabled(true);
    Mixpanel.optInTracking();
    AmplitudeReactNative.setOptOut(false);
  }
  
  static disableTracking() {
    analytics().setAnalyticsCollectionEnabled(false);
    Mixpanel.optOutTracking();
    AmplitudeReactNative.setOptOut(true);
  }
  
  static anonymizeData(data) {
    // Remove PII before tracking
    return {
      ...data,
      email: undefined,
      phone: undefined,
      ip_address: undefined,
      user_id: hashUserId(data.user_id),
    };
  }
  
  static deleteUserData(userId) {
    // GDPR/CCPA compliance
    analytics().deleteAnalyticsData();
    // Additional provider-specific deletion
  }
}

// GDPR/CCPA compliant event tracking
function trackEventWithPrivacy(eventName, properties = {}) {
  if (!PrivacyManager.hasConsent) return;
  
  const anonymizedProperties = PrivacyManager.anonymizeData(properties);
  track(eventName, anonymizedProperties);
}

// Usage with consent
async function initializeAnalytics() {
  const hasConsent = await PrivacyManager.requestConsent();
  
  if (hasConsent) {
    initializeMixpanel('your_token');
    initializeAmplitude('your_api_key');
  }
}

Mini-Project: Add Analytics Event Tracking

Implement a complete analytics system with multiple providers and privacy controls:

// 1. Create unified analytics service
// services/analyticsService.js
import firebase from '@react-native-firebase/app';
import analytics from '@react-native-firebase/analytics';
import Mixpanel from 'react-native-mixpanel';

class AnalyticsService {
  static initialized = false;
  
  static initialize() {
    if (this.initialized) return;
    
    // Initialize providers
    if (!firebase.apps.length) {
      firebase.initializeApp({
        // Your config
      });
    }
    
    Mixpanel.initialize('your_mixpanel_token');
    this.initialized = true;
  }
  
  static track(event, properties = {}) {
    if (!this.initialized) return;
    
    const enrichedProperties = {
      ...properties,
      timestamp: Date.now(),
      platform: Platform.OS,
      app_version: appVersion,
    };
    
    // Track to all providers
    analytics().logEvent(event, enrichedProperties);
    Mixpanel.trackWithProperties(event, enrichedProperties);
  }
  
  static identify(userId, properties = {}) {
    if (!this.initialized) return;
    
    analytics().setUserId(userId);
    Mixpanel.identify(userId);
    Mixpanel.set(properties);
  }
  
  static setUserProperty(key, value) {
    if (!this.initialized) return;
    
    analytics().setUserProperty(key, String(value));
    Mixpanel.set({ [key]: value });
  }
  
  static trackScreen(screenName) {
    this.track('screen_view', {
      screen_name: screenName,
    });
  }
  
  static trackUserAction(action, target, properties = {}) {
    this.track('user_action', {
      action,
      target,
      ...properties,
    });
  }
  
  static trackError(error, context = {}) {
    this.track('app_error', {
      error_message: error.message,
      error_stack: error.stack,
      ...context,
    });
  }
}

// 2. Create React hook
// hooks/useAnalytics.js
import { useCallback } from 'react';
import AnalyticsService from '../services/analyticsService';

export function useAnalytics() {
  const track = useCallback((event, properties) => {
    AnalyticsService.track(event, properties);
  }, []);
  
  const identify = useCallback((userId, properties) => {
    AnalyticsService.identify(userId, properties);
  }, []);
  
  const trackScreen = useCallback((screenName) => {
    AnalyticsService.trackScreen(screenName);
  }, []);
  
  const trackAction = useCallback((action, target, properties) => {
    AnalyticsService.trackUserAction(action, target, properties);
  }, []);
  
  return {
    track,
    identify,
    trackScreen,
    trackAction,
  };
}

// 4. Privacy consent wrapper
// components/AnalyticsProvider.js
function AnalyticsProvider({ children }) {
  const [hasConsent, setHasConsent] = useState(false);
  
  useEffect(() => {
    async function checkConsent() {
      const consent = await AsyncStorage.getItem('analytics_consent');
      setHasConsent(consent === 'true');
    }
    checkConsent();
  }, []);
  
  if (!hasConsent) {
    return (
      <ConsentDialog
        onAccept={() => {
          AsyncStorage.setItem('analytics_consent', 'true');
          setHasConsent(true);
          AnalyticsService.initialize();
        }}
        onDecline={() => {
          AsyncStorage.setItem('analytics_consent', 'false');
        }}
      />
    );
  }
  
  return children;
}
