Why Reanimated?

react-native-reanimated is the modern animation library for React Native. It runs animations entirely on the UI thread, eliminating the bridge bottleneck that causes jank with the built-in Animated API. Reanimated uses worklets — small JavaScript functions that run on the UI thread.

useSharedValue

useSharedValue creates a shared value that can be read and written on both the JS and UI threads. Unlike regular React state, updating a shared value doesn't trigger a re-render — it directly updates the UI thread.

import { useSharedValue } from 'react-native-reanimated';

function MyComponent() {
  // Shared values work on the UI thread
  const translateX = useSharedValue(0);
  const opacity = useSharedValue(1);
  const scale = useSharedValue(1);

  // Modifying shared values
  const handlePress = () => {
    'worklet'; // This function runs on the UI thread
    translateX.value = 100;
    opacity.value = 0.5;
    scale.value = 2;
  };

  return (
    <View>
      <Text>Shared values: {translateX.value}</Text>
    </View>
  );
}

useAnimatedStyle

useAnimatedStyle creates animated styles that update automatically when shared values change. These styles run on the UI thread, making them extremely performant.

import Animated, {
  useSharedValue,
  useAnimatedStyle,
  withTiming,
  withSpring,
} from 'react-native-reanimated';

function AnimatedBox() {
  const translateX = useSharedValue(0);
  const rotate = useSharedValue(0);
  const scale = useSharedValue(1);

  const animatedStyle = useAnimatedStyle(() => {
    'worklet';
    return {
      transform: [
        { translateX: translateX.value },
        { rotate: `${rotate.value}deg` },
        { scale: scale.value },
      ],
    };
  });

  const handleMoveRight = () => {
    'worklet';
    translateX.value = withTiming(200, { duration: 500 });
  };

  const handleRotate = () => {
    'worklet';
    rotate.value = withSpring(rotate.value + 90);
  };

  const handleScale = () => {
    'worklet';
    scale.value = withSpring(scale.value === 1 ? 2 : 1);
  };

  return (
    <View style={styles.container}>
      <Animated.View style={[styles.box, animatedStyle]} />
      <View style={styles.buttons}>
        <Pressable onPress={handleMoveRight} style={styles.button}>
          <Text style={styles.buttonText}>Move Right</Text>
        </Pressable>
        <Pressable onPress={handleRotate} style={styles.button}>
          <Text style={styles.buttonText}>Rotate</Text>
        </Pressable>
        <Pressable onPress={handleScale} style={styles.button}>
          <Text style={styles.buttonText}>Scale</Text>
        </Pressable>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  box: { width: 100, height: 100, backgroundColor: '#3b82f6', borderRadius: 12 },
  buttons: { flexDirection: 'row', gap: 12, marginTop: 32 },
  button: { padding: 10, backgroundColor: '#10b981', borderRadius: 8 },
  buttonText: { color: '#fff', fontWeight: '600' },
});

withSpring and withTiming

Reanimated provides two main animation functions:

  • withTiming: Animates from current value to target over a duration with easing
  • withSpring: Physics-based animation with overshoot and bounce
import Animated, {
  useSharedValue,
  useAnimatedStyle,
  withTiming,
  withSpring,
  withRepeat,
  withSequence,
  Easing,
} from 'react-native-reanimated';

function AnimationExamples() {
  const translateX = useSharedValue(0);
  const opacity = useSharedValue(1);
  const scale = useSharedValue(1);

  // Timing with custom easing
  const moveWithTiming = () => {
    'worklet';
    translateX.value = withTiming(
      translateX.value === 0 ? 200 : 0,
      {
        duration: 800,
        easing: Easing.bezierFn(0.25, 0.1, 0.25, 1),
      }
    );
  };

  // Spring with custom config
  const bounceWithSpring = () => {
    'worklet';
    scale.value = withSpring(1.5, {
      damping: 10,
      stiffness: 100,
      mass: 1,
      velocity: 0,
    });
  };

  // Repeat animation
  const pulseAnimation = () => {
    'worklet';
    opacity.value = withRepeat(
      withSequence(
        withTiming(0.3, { duration: 500 }),
        withTiming(1, { duration: 500 })
      ),
      -1, // Infinite repeat
      true // Reverse on each repeat
    );
  };

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ translateX: translateX.value }, { scale: scale.value }],
    opacity: opacity.value,
  }));

  return (
    <View style={styles.container}>
      <Animated.View style={[styles.box, animatedStyle]} />
      <View style={styles.buttons}>
        <Pressable onPress={moveWithTiming} style={styles.button}>
          <Text style={styles.buttonText}>Timing</Text>
        </Pressable>
        <Pressable onPress={bounceWithSpring} style={styles.button}>
          <Text style={styles.buttonText}>Spring</Text>
        </Pressable>
        <Pressable onPress={pulseAnimation} style={styles.button}>
          <Text style={styles.buttonText}>Pulse</Text>
        </Pressable>
      </View>
    </View>
  );
}

Gesture Animations

Reanimated integrates with react-native-gesture-handler for performant gesture-driven animations. Gestures run entirely on the UI thread.

import { Gesture, GestureDetector } from 'react-native-gesture-handler';
import Animated, {
  useSharedValue,
  useAnimatedStyle,
  withSpring,
} from 'react-native-reanimated';

function DraggableBox() {
  const translateX = useSharedValue(0);
  const translateY = useSharedValue(0);

  const panGesture = Gesture.Pan()
    .onUpdate((e) => {
      'worklet';
      translateX.value = e.changeX;
      translateY.value = e.changeY;
    })
    .onEnd(() => {
      'worklet';
      translateX.value = withSpring(0);
      translateY.value = withSpring(0);
    });

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [
      { translateX: translateX.value },
      { translateY: translateY.value },
    ],
  }));

  return (
    <GestureDetector gesture={panGesture}>
      <Animated.View style={[styles.draggable, animatedStyle]} />
    </GestureDetector>
  );
}

// Pinch gesture for scaling
function PinchableBox() {
  const scale = useSharedValue(1);

  const pinchGesture = Gesture.Pinch()
    .onUpdate((e) => {
      'worklet';
      scale.value = e.scale;
    })
    .onEnd(() => {
      'worklet';
      scale.value = withSpring(1);
    });

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: scale.value }],
  }));

  return (
    <GestureDetector gesture={pinchGesture}>
      <Animated.View style={[styles.box, animatedStyle]} />
    </GestureDetector>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  box: { width: 120, height: 120, backgroundColor: '#8b5cf6', borderRadius: 12 },
  draggable: { width: 100, height: 100, backgroundColor: '#ef4444', borderRadius: 12 },
  buttons: { flexDirection: 'row', gap: 12, marginTop: 32 },
  button: { padding: 10, backgroundColor: '#3b82f6', borderRadius: 8 },
  buttonText: { color: '#fff', fontWeight: '600' },
});

Layout Animations

Reanimated provides layout animations that automatically animate elements when they enter, exit, or move within a layout.

import Animated, {
  FadeIn,
  FadeOut,
  SlideInRight,
  Layout,
  ZoomIn,
} from 'react-native-reanimated';

function AnimatedList({ items }) {
  return (
    <View>
      {items.map((item, index) => (
        <Animated.View
          key={item.id}
          layout={Layout.springify()}  // Animate layout changes
          entering={SlideInRight.delay(index * 100).springify()}  // Entry animation
          exiting={FadeOut.duration(300)}  // Exit animation
          style={styles.listItem}
        >
          <Text style={styles.listText}>{item.text}</Text>
        </Animated.View>
      ))}
    </View>
  );
}

// Available entering/exiting animations:
// FadeIn, FadeInDown, FadeInUp, FadeInLeft, FadeInRight
// FadeOut, FadeOutDown, FadeOutUp, FadeOutLeft, FadeOutRight
// SlideInDown, SlideInUp, SlideInLeft, SlideInRight
// SlideOutDown, SlideOutUp, SlideOutLeft, SlideOutRight
// ZoomIn, ZoomInRotate, ZoomOut, ZoomOutRotate
// BounceIn, BounceOut

const styles = StyleSheet.create({
  listItem: { padding: 16, backgroundColor: '#fff', borderRadius: 8, marginBottom: 8 },
  listText: { fontSize: 16 },
});

Mini-Project: Build an Animated Box That Moves on Press

Create a draggable box with spring-back animation:

// components/AnimatedBox.js
import { StyleSheet, View } from 'react-native';
import { Gesture, GestureDetector } from 'react-native-gesture-handler';
import Animated, {
  useSharedValue,
  useAnimatedStyle,
  withSpring,
  withTiming,
} from 'react-native-reanimated';

export default function AnimatedBox() {
  const translateX = useSharedValue(0);
  const translateY = useSharedValue(0);
  const scale = useSharedValue(1);

  const panGesture = Gesture.Pan()
    .onBegin(() => {
      'worklet';
      scale.value = withTiming(1.1, { duration: 200 });
    })
    .onUpdate((e) => {
      'worklet';
      translateX.value = e.changeX;
      translateY.value = e.changeY;
    })
    .onEnd(() => {
      'worklet';
      translateX.value = withSpring(0, { damping: 15 });
      translateY.value = withSpring(0, { damping: 15 });
      scale.value = withSpring(1);
    });

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [
      { translateX: translateX.value },
      { translateY: translateY.value },
      { scale: scale.value },
    ],
  }));

  return (
    <View style={styles.container}>
      <GestureDetector gesture={panGesture}>
        <Animated.View style={[styles.box, animatedStyle]}>
          <Animated.Text style={styles.text}>Drag me!</Animated.Text>
        </Animated.View>
      </GestureDetector>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  box: {
    width: 150,
    height: 150,
    backgroundColor: '#3b82f6',
    borderRadius: 16,
    justifyContent: 'center',
    alignItems: 'center',
    elevation: 8,
    shadowColor: '#000',
    shadowOffset: { width: 0, height: 4 },
    shadowOpacity: 0.2,
    shadowRadius: 8,
  },
  text: { color: '#fff', fontSize: 16, fontWeight: '600' },
});

Summary

  • useSharedValue creates values that update on the UI thread without re-renders
  • useAnimatedStyle binds shared values to component styles
  • withTiming provides easing-based animations with custom durations
  • withSpring creates physics-based animations with natural bounce
  • Gestures from react-native-gesture-handler run entirely on the UI thread
  • Layout animations (FadeIn, SlideInRight, etc.) animate element entry/exit automatically
  • Always add the 'worklet' directive to functions that run on the UI thread
