What is SQLite in React Native?

SQLite is a lightweight, relational database that runs locally on the device. It's ideal for apps that need structured data storage with complex queries, relationships between data, and ACID transactions.

When to use SQLite vs key-value storage?

  • Use SQLite when: You need complex queries, joins, relationships, or structured data
  • Use MMKV/AsyncStorage when: You need simple key-value pairs like settings or tokens
  • SQLite handles: Offline-first apps, chat messages, notes, inventory systems

Installation

npx expo install expo-sqlite

Opening a Database

Use openDatabaseAsync to create or open a database. The database persists across app launches.

// database/db.js
import * as SQLite from 'expo-sqlite';

let db = null;

export async function getDatabase() {
  if (!db) {
    db = await SQLite.openDatabaseAsync('myapp.db');
  }
  return db;
}

// Initialize tables on app start
export async function initDatabase() {
  const database = await getDatabase();

  await database.execAsync(`
    PRAGMA journal_mode = WAL;
    PRAGMA foreign_keys = ON;

    CREATE TABLE IF NOT EXISTS notes (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      content TEXT,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
      updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
    );
  `);

  console.log('Database initialized');
}

CREATE TABLE

Define your table schema using standard SQL. Always use IF NOT EXISTS to avoid errors on subsequent launches.

// Create a users table
await database.execAsync(`
  CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE NOT NULL,
    email TEXT UNIQUE NOT NULL,
    avatar TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
  );
`);

// Create a posts table with foreign key
await database.execAsync(`
  CREATE TABLE IF NOT EXISTS posts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    title TEXT NOT NULL,
    body TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
  );
`);

INSERT with Parameterized Queries

Always use parameterized queries to prevent SQL injection. Never concatenate user input directly into SQL strings.

// Insert a single note
const result = await database.runAsync(
  'INSERT INTO notes (title, content) VALUES (?, ?)',
  ['My First Note', 'This is the content of my note']
);
console.log('Inserted note with ID:', result.lastInsertRowId);

// Insert multiple notes
const notes = [
  { title: 'Shopping List', content: 'Milk, Eggs, Bread' },
  { title: 'Meeting Notes', content: 'Discuss Q4 targets' },
  { title: 'Ideas', content: 'Build a recipe app' },
];

for (const note of notes) {
  await database.runAsync(
    'INSERT INTO notes (title, content) VALUES (?, ?)',
    [note.title, note.content]
  );
}

// Insert with named parameters
await database.runAsync(
  'INSERT INTO users (username, email) VALUES ($username, $email)',
  { $username: 'johndoe', $email: 'john@example.com' }
);

SELECT Queries

Use getAllAsync for multiple rows and getFirstAsync for a single row.

// Get all notes
const allNotes = await database.getAllAsync('SELECT * FROM notes');
console.log(allNotes);
// [{ id: 1, title: 'My First Note', content: '...', created_at: '...' }]

// Get notes with a condition
const filtered = await database.getAllAsync(
  'SELECT * FROM notes WHERE title LIKE ?',
  ['%Shopping%']
);

// Get a single note by ID
const note = await database.getFirstAsync(
  'SELECT * FROM notes WHERE id = ?',
  [1]
);

// Get notes with JOIN
const notesWithUsers = await database.getAllAsync(`
  SELECT notes.*, users.username
  FROM notes
  JOIN users ON notes.user_id = users.id
  ORDER BY notes.created_at DESC
`);

// Aggregate queries
const stats = await database.getFirstAsync(
  'SELECT COUNT(*) as total, MIN(created_at) as first_note FROM notes'
);

UPDATE and DELETE

// Update a note
await database.runAsync(
  'UPDATE notes SET title = ?, content = ?, updated_at = CURRENT_TIMESTAMP WHERE id = ?',
  ['Updated Title', 'Updated content', 1]
);

// Delete a note
await database.runAsync('DELETE FROM notes WHERE id = ?', [1]);

// Delete all notes
await database.runAsync('DELETE FROM notes');

// Check rows affected
const result = await database.runAsync(
  'UPDATE notes SET title = ? WHERE id = ?',
  ['New Title', 999]
);
console.log('Rows affected:', result.changes);

Database Migrations

Use a version table to track schema changes and run migrations when the app updates.

// migrations/migrate.js
export async function migrate(database) {
  // Create migrations table
  await database.execAsync(`
    CREATE TABLE IF NOT EXISTS migrations (
      version INTEGER PRIMARY KEY,
      applied_at DATETIME DEFAULT CURRENT_TIMESTAMP
    );
  `);

  const current = await database.getFirstAsync(
    'SELECT MAX(version) as version FROM migrations'
  );
  const currentVersion = current?.version || 0;

  const migrations = [
    {
      version: 1,
      up: `CREATE TABLE IF NOT EXISTS notes (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        title TEXT NOT NULL,
        content TEXT,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP
      );`,
    },
    {
      version: 2,
      up: `ALTER TABLE notes ADD COLUMN updated_at DATETIME DEFAULT CURRENT_TIMESTAMP;`,
    },
    {
      version: 3,
      up: `CREATE TABLE IF NOT EXISTS tags (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT UNIQUE NOT NULL
      );`,
    },
  ];

  for (const migration of migrations) {
    if (migration.version > currentVersion) {
      await database.execAsync(migration.up);
      await database.runAsync(
        'INSERT INTO migrations (version) VALUES (?)',
        [migration.version]
      );
      console.log(`Applied migration ${migration.version}`);
    }
  }
}

Summary

  • expo-sqlite provides local SQLite database support for React Native
  • Use parameterized queries (?) to prevent SQL injection
  • getAllAsync returns multiple rows; getFirstAsync returns one row
  • Use migrations with a version table to manage schema changes
  • Enable WAL mode for better performance
  • SQLite is ideal for structured, relational data with complex queries
